48/** Class that contains information about an inheritance relation.
49 */
50struct BaseClassDef
51{
52 BaseClassDef(ClassDef *cd,const DString &n,Protection p, Specifier v,const DString &t) :
53 classDef(cd), usedName(n), prot(p), virt(v), templSpecifiers(t) {}
54
55 /** Class definition that this relation inherits from. */
56 ClassDef *classDef;
57
58 /** name used in the inheritance list
59 * (may be a typedef name instead of the class name)
60 */
61 DString usedName;
62
63 /** Protection level of the inheritance relation:
64 * Public, Protected, or Private
65 */
66 Protection prot;
67
68 /** Virtualness of the inheritance relation:
69 * Normal, or Virtual
70 */
71 Specifier virt;
72
73 /** Template arguments used for the base class */
74 DString templSpecifiers;
75};

473/** Class that contains information about a usage relation.
474 */
475struct UsesClassDef
476{
477 UsesClassDef(ClassDef *cd) : classDef(cd) {}
478 void addAccessor(const DString &s)
479 {
480 if (accessors.find(s.str())==accessors.end())
481 {
482 accessors.insert(s.str());
483 }
484 }
485 /** Class definition that this relation uses. */
486 ClassDef *classDef;
487
488 /** Dictionary of member variable names that form the edge labels of the
489 * usage relation.
490 */
491 StringSet accessors;
492
493 /** Template arguments used for the base class */
494 DString templSpecifiers;
495
496 bool containment = true;
497};
498
499class UsesClassList final : public std::vector<UsesClassDef>
500{
501};
