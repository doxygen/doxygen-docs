62const char *LatexDocVisitor::getSectionName(int level) const
63{
64 bool compactLatex = Config_getBool(COMPACT_LATEX);
65 int l = level;
66 if (compactLatex) l++;
67
68 if (l < g_maxLevels)
69 {
70 l += m_hierarchyLevel; /* May be -1 if generating main page */
71 // Sections get special treatment because they inherit the parent's level
72 if (l >= g_maxLevels)
73 {
74 l = g_maxLevels - 1;
75 }
76 else if (l < 0)
77 {
78 /* Should not happen; level is always >= 1 and hierarchyLevel >= -1 */
79 l = 0;
80 }
81 return g_secLabels[l];
82 }
83 else if (l == 7)
84 {
85 return g_paragraphLabel;
86 }
87 else
88 {
89 return g_subparagraphLabel;
90 }

93static void insertDimension(TextStream &t, DString dimension, const char *orientationString)
94{
95 // dimensions for latex images can be a percentage, in this case they need some extra
96 // handling as the % symbol is used for comments
97 static const reg::Ex re(R"((\d+)%)");
98 std::string s = dimension.str();
99 reg::Match match;
100 if (reg::search(s,match,re))
101 {
102 bool ok = false;
103 double percent = DString(match[1].str()).toInt(&ok);
104 if (ok)
105 {
106 t << percent/100.0 << "\\text" << orientationString;
107 return;
108 }
109 }
110 t << dimension;
111}

113static void visitPreStart(TextStream &t, bool hasCaption, DString name, DString width, DString height, bool inlineImage = false)
114{
115 if (inlineImage)
116 {
117 t << "\n\\begin{DoxyInlineImage}%\n";
118 }
119 else
120 {
121 if (hasCaption)
122 {
123 t << "\n\\begin{DoxyImage}%\n";
124 }
125 else
126 {
127 t << "\n\\begin{DoxyImageNoCaption}%\n"
128 " \\doxymbox{";
129 }
130 }
131
132 t << "\\includegraphics";
133 if (!width.empty() || !height.empty())
134 {
135 t << "[";
136 }
137 if (!width.empty())
138 {
139 t << "width=";
140 insertDimension(t, width, "width");
141 }
142 if (!width.empty() && !height.empty())
143 {
144 t << ",";
145 }
146 if (!height.empty())
147 {
148 t << "height=";
149 insertDimension(t, height, "height");
150 }
151 if (width.empty() && height.empty())
152 {
153 /* default setting */
154 if (inlineImage)
155 {
156 t << "[height=\\baselineskip,keepaspectratio=true]";
157 }
158 else
159 {
160 t << "[width=\\textwidth,height=\\textheight/2,keepaspectratio=true]";
161 }
162 }
163 else
164 {
165 t << "]";
166 }
167
168 t << "{" << name << "}";
169
170 if (hasCaption)
171 {
172 if (!inlineImage)
173 {
174 if (Config_getBool(PDF_HYPERLINKS))
175 {
176 t << "%\n\\doxyfigcaption{";
177 }
178 else
179 {
180 t << "%\n\\doxyfigcaptionnolink{";
181 }
182 }
183 else
184 {
185 t << "%"; // to catch the caption
186 }
187 }
188}

269 {

271 }
272 else

274 m_t << "$<$";
275 }
276 }
277 else if (((s.symbol() == HtmlEntityMapper::Sym_gt) || (s.symbol() == HtmlEntityMapper::Sym_Greater)) && (!m_insidePre))
278 {
279 if (pdfHyperlinks)
280 {
281 m_t << "\\texorpdfstring{$>$}{>}";
282 }
283 else
284 {
285 m_t << "$>$";
286 }
287 }
288 else
289 {
290 m_t << res;
291 }
292 }
293 else
294 {
295 err("LaTeX: non supported HTML-entity found: {}\n",HtmlEntityMapper::instance().html(s.symbol(),true));

409void LatexDocVisitor::operator()(const DocVerbatim &s)
410{
411 if (m_hide) return;
412 DString lang = m_langExt;
413 if (!s.language().empty()) // explicit language setting
414 {
415 lang = s.language();
416 }
417 SrcLangExt langExt = getLanguageFromCodeLang(lang);
418 switch(s.type())
419 {
420 case DocVerbatim::Code:
421 {
422 m_ci.startCodeFragment("DoxyCode");
423 getCodeParser(lang).parseCode(m_ci,s.context(),s.text(),langExt,
424 Config_getBool(STRIP_CODE_COMMENTS),
425 CodeParserOptions().setExample(s.isExample(),s.exampleFile()));
426 m_ci.endCodeFragment("DoxyCode");
427 }
428 break;
429 case DocVerbatim::JavaDocLiteral:
430 filter(s.text(), true);
431 break;
432 case DocVerbatim::JavaDocCode:
433 m_t << "{\\ttfamily ";
434 filter(s.text(), true);
435 m_t << "}";
436 break;
437 case DocVerbatim::Verbatim:
438 if (isTableNested(s.parent())) // in table
439 {
440 m_t << "\\begin{DoxyCode}{0}";
441 filter(s.text(), true);
442 m_t << "\\end{DoxyCode}\n";
443 }
444 else
445 {
446 m_t << "\\begin{DoxyVerb}";
447 m_t << s.text();
448 m_t << "\\end{DoxyVerb}\n";
449 }
450 break;
451 case DocVerbatim::HtmlOnly:
452 case DocVerbatim::XmlOnly:
453 case DocVerbatim::ManOnly:
454 case DocVerbatim::RtfOnly:
455 case DocVerbatim::DocbookOnly:
456 /* nothing */
457 break;
458 case DocVerbatim::LatexOnly:
459 m_t << s.text();
460 break;
461 case DocVerbatim::Dot:
462 {
463 bool exists = false;
464 auto fileName = writeInlineGraph(Config_getString(LATEX_OUTPUT)+"/inline_dotgraph_", // baseName
465 ".dot", // extension
466 s.text(), // contents
467 exists);
468 if (!fileName.empty())
469 {
470 startDotFile(fileName,s.width(),s.height(),s.hasCaption(),s.srcFile(),s.srcLine(),!exists);
471 visitChildren(s);
472 endDotFile(s.hasCaption());
473 }
474 }
475 break;
476 case DocVerbatim::Msc:
477 {
478 bool exists = false;
479 auto fileName = writeInlineGraph(Config_getString(LATEX_OUTPUT)+"/inline_mscgraph_", // baseName
480 ".msc", // extension
481 "msc {"+s.text()+"}", // contents
482 exists);
483 if (!fileName.empty())
484 {
485 writeMscFile(fileName, s, !exists);
486 }
487 }
488 break;
489 case DocVerbatim::PlantUML:
490 {
491 DString latexOutput = Config_getString(LATEX_OUTPUT);
492 auto baseNameVector = PlantumlManager::instance().writePlantUMLSource(
493 latexOutput,s.exampleFile(),s.text(),
494 s.useBitmap() ? PlantumlManager::PUML_BITMAP : PlantumlManager::PUML_EPS,
495 s.engine(),s.srcFile(),s.srcLine(),true);
496
497 for (const auto &baseName: baseNameVector)
498 {
499 writePlantUMLFile(baseName, s);
500 }
501 }
502 break;
503 case DocVerbatim::Mermaid:
504 if (Config_getBool(MERMAID_RENDER_MODE)!=MERMAID_RENDER_MODE_t::CLIENT_SIDE)
505 {
506 auto latexOutput = Config_getString(LATEX_OUTPUT);
507 auto outputFormat = MermaidManager::OutputFormat::LaTeX;
508 auto imageFormat = MermaidManager::convertToImageFormat(outputFormat);
509 DString baseName = MermaidManager::instance().writeMermaidSource(
510 latexOutput,s.exampleFile(),s.text(),imageFormat,
511 s.srcFile(),s.srcLine());
512 writeMermaidFile(baseName, s);
513 }
514 break;
515 }
516}

529void LatexDocVisitor::operator()(const DocInclude &inc)
530{
531 if (m_hide) return;
532 SrcLangExt langExt = getLanguageFromFileName(inc.extension());
533 switch(inc.type())
534 {
535 case DocInclude::IncWithLines:
536 {
537 m_ci.startCodeFragment("DoxyCodeInclude");
538 FileInfo cfi( inc.file().str() );
539 auto fd = createFileDef( cfi.dirPath(), cfi.fileName() );
540 getCodeParser(inc.extension()).parseCode(m_ci,inc.context(),
541 inc.text(),
542 langExt,
543 inc.stripCodeComments(),
544 CodeParserOptions()
545 .setExample(inc.isExample(), inc.exampleFile())
546 .setFileDef(fd.get())
547 .setInlineFragment(true)
548 );
549 m_ci.endCodeFragment("DoxyCodeInclude");
550 }
551 break;
552 case DocInclude::Include:
553 {
554 m_ci.startCodeFragment("DoxyCodeInclude");
555 getCodeParser(inc.extension()).parseCode(m_ci,inc.context(),
556 inc.text(),langExt,
557 inc.stripCodeComments(),
558 CodeParserOptions()
559 .setExample(inc.isExample(), inc.exampleFile())
560 .setInlineFragment(true)
561 .setShowLineNumbers(false)
562 );
563 m_ci.endCodeFragment("DoxyCodeInclude");
564 }
565 break;
566 case DocInclude::DontInclude:
567 case DocInclude::DontIncWithLines:
568 case DocInclude::HtmlInclude:
569 case DocInclude::RtfInclude:
570 case DocInclude::ManInclude:
571 case DocInclude::XmlInclude:
572 case DocInclude::DocbookInclude:
573 break;
574 case DocInclude::LatexInclude:
575 m_t << inc.text();
576 break;
577 case DocInclude::VerbInclude:
578 if (isTableNested(inc.parent())) // in table
579 {
580 m_t << "\\begin{DoxyCode}{0}";
581 filter(inc.text(), true);
582 m_t << "\\end{DoxyCode}\n";
583 }
584 else
585 {
586 m_t << "\n\\begin{DoxyVerbInclude}\n";
587 m_t << inc.text();
588 m_t << "\\end{DoxyVerbInclude}\n";
589 }
590 break;
591 case DocInclude::Snippet:
592 case DocInclude::SnippetWithLines:
593 {
594 m_ci.startCodeFragment("DoxyCodeInclude");
595 CodeFragmentManager::instance().parseCodeFragment(m_ci,
596 inc.file(),
597 inc.blockId(),
598 inc.context(),
599 inc.type()==DocInclude::SnippetWithLines,
600 inc.trimLeft(),
601 inc.stripCodeComments()
602 );
603 m_ci.endCodeFragment("DoxyCodeInclude");
604 }
605 break;
606 }

609void LatexDocVisitor::operator()(const DocIncOperator &op)
610{
611 //printf("DocIncOperator: type=%d first=%d, last=%d text='%s'\n",
612 // op.type(),op.isFirst(),op.isLast(),qPrint(op.text()));
613 if (op.isFirst())
614 {
615 if (!m_hide) m_ci.startCodeFragment("DoxyCodeInclude");
616 pushHidden(m_hide);
617 m_hide = true;
618 }
619 DString locLangExt = getFileNameExtension(op.includeFileName());
620 if (locLangExt.empty()) locLangExt = m_langExt;
621 SrcLangExt langExt = getLanguageFromFileName(locLangExt);
622 if (op.type()!=DocIncOperator::Skip)
623 {
624 m_hide = popHidden();
625 if (!m_hide)
626 {
627 std::unique_ptr<FileDef> fd;
628 if (!op.includeFileName().empty())
629 {
630 FileInfo cfi( op.includeFileName().str() );
631 fd = createFileDef( cfi.dirPath(), cfi.fileName() );
632 }
633
634 getCodeParser(locLangExt).parseCode(m_ci,op.context(),op.text(),langExt,
635 op.stripCodeComments(),
636 CodeParserOptions()
637 .setExample(op.isExample(),op.exampleFile())
638 .setFileDef(fd.get())
639 .setStartLine(op.line())
640 .setShowLineNumbers(op.showLineNo())
641 );
642 }
643 pushHidden(m_hide);
644 m_hide=true;
645 }
646 if (op.isLast())
647 {
648 m_hide=popHidden();
649 if (!m_hide) m_ci.endCodeFragment("DoxyCodeInclude");
650 }
651 else
652 {
653 if (!m_hide) m_t << "\n";
654 }
655}

687{
688 if (m_hide) return;
689 auto opt = cite.option();
690 DString txt;
691 if (opt.noCite())
692 {
693 if (!cite.file().empty())
694 {
695 txt = cite.getText();
696 }
697 else
698 {
699 if (!opt.noPar()) txt += "[";
700 txt += cite.target();
701 if (!opt.noPar()) txt += "]";
702 }
703 m_t << "{\\bfseries ";
704 filter(txt);
705 m_t << "}";
706 }
707 else
708 {
709 if (!cite.file().empty())
710 {
711 DString anchor = cite.anchor();
712 DString anchorPrefix = CitationManager::instance().anchorPrefix();
713 anchor = anchor.mid(anchorPrefix.length()); // strip prefix
714
715 txt = "\\DoxyCite{" + anchor + "}";
716 if (opt.isNumber())
717 {
718 txt += "{number}";
719 }
720 else if (opt.isShortAuthor())
721 {
722 txt += "{shortauthor}";
723 }
724 else if (opt.isYear())
725 {
726 txt += "{year}";
727 }
728 if (!opt.noPar()) txt += "{1}";
729 else txt += "{0}";
730
731 m_t << txt;
732 }
733 else
734 {
735 if (!opt.noPar()) txt += "[";
736 txt += cite.target();
737 if (!opt.noPar()) txt += "]";
738 m_t << "{\\bfseries ";
739 filter(txt);
740 m_t << "}";
741 }
742 }

1310
1311 setCurrentColumn(currentColumn()+1);
1312
1313 DString cellOpts;
1314 DString cellSpec;
1315 auto appendOpt = [&cellOpts](const DString &s)
1316 {
1317 if (!cellOpts.empty()) cellOpts+=",";
1318 cellOpts+=s;
1319 };
1320 auto appendSpec = [&cellSpec](const DString &s)
1321 {
1322 if (!cellSpec.empty()) cellSpec+=",";
1323 cellSpec+=s;
1324 };

1326 {
1327 if (!cellOpts.empty() || !cellSpec.empty())

1329 m_t << "\\SetCell";
1330 if (!cellOpts.empty())
1331 {
1332 m_t << "[" << cellOpts << "]";
1333 }
1334 m_t << "{" << cellSpec << "}";
1335 }
1336 };
1337
1338 // skip over columns that have a row span starting at an earlier row
1339 for (const auto &span : rowSpans())
1340 {
1341 //printf("span(r=%u,c=%u): column=%zu colSpan=%zu,rowSpan=%zu currentColumn()=%zu\n",
1342 // span.cell.rowIndex(),span.cell.columnIndex(),
1343 // span.column,span.colSpan,span.rowSpan,
1344 // currentColumn());
1345 if (span.rowSpan>0 && span.column==currentColumn())

1349 {
1350 m_t << "&";
1351 }

1371 }
