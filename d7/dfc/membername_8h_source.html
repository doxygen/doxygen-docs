59/** Ordered dictionary of MemberName objects. */
60class MemberNameLinkedMap final : public LinkedMap<MemberName>
61{
62 public:
63 MemberName::Ptr take(const DString &key,const MemberDef *value)
64 {
65 MemberName::Ptr result;
66 MemberName *mn = find(key);
67 if (mn)
68 {
69 auto it = std::find_if(mn->begin(),mn->end(),[&value](const auto &el) { return el.get()==value; });
70 if (it != mn->end())
71 {
72 it->swap(result);
73 mn->erase(it);
74 }
75 if (mn->empty())
76 {
77 del(key);
78 }
79 }
80 return result;
81 }
82};
