59/** \brief A list of index interfaces.
60 *
61 * This class itself implements all methods of IndexIntf and
62 * just forwards the calls to all items in the list (composite design pattern).
63 */
64class IndexList
65{
66 using IndexPtr = std::unique_ptr<IndexIntf>;
67
68 template<class... Ts,class... As>
69 void foreach(void (IndexIntf::*methodPtr)(Ts...),As&&... args)
70 {
71 for (const auto &intf : m_indices)
72 {
73 (intf.get()->*methodPtr)(std::forward<As>(args)...);
74 }
75 }
76
77 template<class... Ts,class... As>
78 void foreach_locked(void (IndexIntf::*methodPtr)(Ts...),As&&... args)
79 {
80 std::lock_guard<std::mutex> lock(m_mutex);
81 for (const auto &intf : m_indices)
82 {
83 (intf.get()->*methodPtr)(std::forward<As>(args)...);
84 }
85 }
86
87 public:
88 /** disable the indices */
89 void disable()
90 { m_enabled = false; }
91
92 /** enable the indices */
93 void enable()
94 { m_enabled = true; }
95
96 /** Add an index generator to the list, using a syntax similar to std::make_unique<T>() */
97 template<class T,class... As>
98 void addIndex(As&&... args)
99 { m_indices.push_back(std::make_unique<T>(std::forward<As>(args)...)); }
100
101 void initialize()
102 { foreach(&IndexIntf::initialize); }
103
104 void finalize()
105 { foreach(&IndexIntf::finalize); }
106
107 void incContentsDepth()
108 { if (m_enabled) foreach_locked(&IndexIntf::incContentsDepth); }
109
110 void decContentsDepth()
111 { if (m_enabled) foreach_locked(&IndexIntf::decContentsDepth); }
112
113 void addContentsItem(bool isDir, const DString &name, const DString &ref,
114 const DString &file, const DString &anchor,bool separateIndex=false,bool addToNavIndex=false,
115 const Definition *def=nullptr, const DString &nameAsHtml = DString())
116 { if (m_enabled) foreach_locked(&IndexIntf::addContentsItem,isDir,name,ref,file,anchor,separateIndex,addToNavIndex,def,nameAsHtml); }
117
118 void addIndexItem(const Definition *context,const MemberDef *md,const DString &sectionAnchor=DString(),const DString &title=DString())
119 { if (m_enabled) foreach_locked(&IndexIntf::addIndexItem,context,md,sectionAnchor,title); }
120
121 void addIndexFile(const DString &name)
122 { if (m_enabled) foreach_locked(&IndexIntf::addIndexFile,name); }
123
124 void addImageFile(const DString &name)
125 { if (m_enabled) foreach_locked(&IndexIntf::addImageFile,name); }
126
127 void addStyleSheetFile(const DString &name)
128 { if (m_enabled) foreach_locked(&IndexIntf::addStyleSheetFile,name); }
129
130 private:
131 bool m_enabled = true;
132 std::mutex m_mutex;
133 std::vector<IndexPtr> m_indices;
134};
