54 void sortReferences()
55 {
56 auto comp = [](const Definition *c1,const Definition *c2)
57 {
58 return Config_getBool(SORT_BY_SCOPE_NAME) ?
59 dstricmp_sort(c1->qualifiedName(), c2->qualifiedName())<0 :
60 dstricmp_sort(c1->name(), c2->name())<0;
61 };
62 std::stable_sort(m_satisfiedBy.begin(), m_satisfiedBy.end(), comp);
63 std::stable_sort(m_verifiedBy.begin(), m_verifiedBy.end(), comp);
64 }
65
66 void addSatisfiedBy(const Definition *def)
67 {
68 if (auto it = std::find(m_satisfiedBy.begin(),m_satisfiedBy.end(),def); it==m_satisfiedBy.end())
69 {
70 m_satisfiedBy.push_back(def);
71 }
72 }
73
74 void addVerifiedBy(const Definition *def)
75 {
76 if (auto it = std::find(m_verifiedBy.begin(),m_verifiedBy.end(),def); it==m_verifiedBy.end())
77 {
78 m_verifiedBy.push_back(def);
79 }
80 }

116void RequirementManager::addRequirement(Entry *e)
117{
118 DString tagFile;
119 DString extPage;
120 if (!Config_getBool(GENERATE_REQUIREMENTS)) // still record the requirement, so we can refer to it.
121 {
122 if (e->tagInfo())
123 {
124 //printf("External requirement %s title=%s fileName=%s tagName=%s anchor=%s\n",
125 // qPrint(e->name),qPrint(e->type),qPrint(e->tagInfo()->fileName),qPrint(e->tagInfo()->tagName),qPrint(e->tagInfo()->anchor));
126 tagFile = e->tagInfo()->tagName;
127 extPage = e->tagInfo()->fileName;
128 // register requirement id as anchor; for non-external links this is
129 // done in commentscan.l in the comment block containing the @requirement command
130 SectionManager::instance().add(e->name,"requirements",1,e->type,SectionType::Requirement,1);
131 }
132 p->requirements.add(e->name, e->fileName, e->startLine, e->type, e->doc, tagFile, extPage);
133 }
134 else
135 {
136 if (p->reqPageDef==nullptr)
137 {
138 p->reqPageDef = addRelatedPage("requirements", // name
139 theTranslator->trRequirements(), // ptitle
140 DString(), // doc
141 "requirements", // fileName
142 1, // docLine
143 1 // startLine
144 );
145 }
146 //printf("requirement ID=%s title='%s' file=%s line=%d brief='%s' doc='%s'\n",
147 // qPrint(e->name), qPrint(e->type), qPrint(e->fileName), e->startLine, qPrint(e->brief), qPrint(e->doc));
148 DString title = parseCommentAsText(p->reqPageDef,nullptr,e->type,e->fileName,e->startLine);
149 DString doc = e->doc;
150 if (e->tagInfo())
151 {
152 //printf("External requirement %s title=%s fileName=%s tagName=%s anchor=%s\n",
153 // qPrint(e->name),qPrint(e->type),qPrint(e->tagInfo()->fileName),qPrint(e->tagInfo()->tagName),qPrint(e->tagInfo()->anchor));
154 tagFile = e->tagInfo()->tagName;
155 extPage = e->tagInfo()->fileName;
156 // register requirement id as anchor; for non-external links this is
157 // done in commentscan.l in the comment block containing the @requirement command
158 SectionManager::instance().add(e->name,"requirements",1,title,SectionType::Anchor,1);
159 }
160 if (!e->brief.empty())
161 {
162 doc.prepend(e->brief+"\n<p>");
163 }
164 p->requirements.add(e->name, e->fileName, e->startLine, title, doc, tagFile, extPage);
165
166 p->reqPageDef->setRefItems(e->sli);
167 }

185void RequirementManager::generatePage()
186{
187 AUTO_TRACE("#requirements={}",p->requirements.size());
188 if (!Config_getBool(GENERATE_REQUIREMENTS) || p->requirements.empty()) return;
189 std::vector<const SectionInfo*> anchors;
190 std::stable_sort(p->requirements.begin(),p->requirements.end(),
191 [](const auto &left,const auto &right) { return dstricmp(left->id(),right->id()) < 0; });
192 DString doc = "<table class=\"doxtable reqlist\">";
193 doc.reserve(10*1024); // prevent too many reallocs
194 doc += "<tr><th>";
195 doc += theTranslator->trRequirementID();
196 doc += "</th><th>";
197 doc += "</th>";
198 using RequirementPtrVector = std::vector<const Requirement*>;
199 RequirementPtrVector missingSatisfiedRef, missingVerifiedRef;
200 for (const auto &req : p->requirements) // TODO: filter out external references?
201 {
202 if (const SectionInfo *si = SectionManager::instance().find(req->id()); si!=nullptr)
203 {
204 anchors.push_back(si);
205 }
206 doc += "<tr><td valign=\"top\">";
207 doc += " \\ifile \""+req->file()+"\" \\iline "+DString().setNum(req->line())+" ";
208 doc += "\\anchor ";
209 doc += req->id();
210 doc += " ";
211 doc += "<span class=\"req_id\">";
212 if (DString tagFile = req->getTagFile(); !tagFile.empty())
213 {
214 //printf("tagFile=%s extPage=%s\n",qPrint(tagFile),qPrint(req->getExtPage()));
215 doc += "<a href=\"";
216 doc += createHtmlUrl(DString(),tagFile,false,req->getExtPage(),req->id());
217 doc +="\">"+req->id()+"</a>";
218 }
219 else
220 {
221 doc += req->id();
222 }
223 doc += "</span> ";
224 doc += "</td><td>";
225 doc += "<div class=\"req_title\">"+req->title()+"</div>";
226 doc += "<p/><div class=\"req_docs\">";
227 doc += req->doc();
228 req->sortReferences();
229 auto symToString = [](const Definition *sym)
230 {
231 DString symName = sym->qualifiedName();
232 if (sym->definitionType()==Definition::TypeMember)
233 {
234 const MemberDef *md = toMemberDef(sym);
235 if (!md->isObjCMethod() && md->isFunctionOrSignalSlot()) symName += "()";
236 }
237 return symName;
238 };
239 int numSatisfiedBy = static_cast<int>(req->satisfiedBy().size());
240 if (numSatisfiedBy>0)
241 {
242 doc += "<p><div class=\"satisfiedby\">";
243 doc += theTranslator->trSatisfiedBy(
244 writeMarkerList(theTranslator->trWriteList(numSatisfiedBy).str(),
245 numSatisfiedBy,
246 [&symToString, &refs = req->satisfiedBy()](size_t entryIndex) {
247 return symToString(refs[entryIndex]);
248 }));
249 doc += "</div></p>";
250 }
251 else
252 {
253 missingSatisfiedRef.push_back(req.get());
254 }
255 int numVerifiedBy = static_cast<int>(req->verifiedBy().size());
256 if (numVerifiedBy>0)
257 {
258 doc += "<p><div class=\"verifiedby\">";
259 doc += theTranslator->trVerifiedBy(
260 writeMarkerList(theTranslator->trWriteList(numVerifiedBy).str(),
261 numVerifiedBy,
262 [&symToString, &refs = req->verifiedBy()](size_t entryIndex) {
263 return symToString(refs[entryIndex]);
264 }));
265 doc += "</div></p>";
266 }
267 else
268 {
269 missingVerifiedRef.push_back(req.get());
270 }
271 doc += "\n</div></td></tr>\n";
272 }
273 doc += "</table>\n";
274
275 //------------
276 doc += " \\ifile \"requirements\" \\iline 1\n";

278 auto writeMissingRef = [&doc,&anchors](const RequirementPtrVector &reqs,
279 const char *label,const DString &section,const DString &text)
280 {
281 if (!reqs.empty())
282 {
283 SectionManager &sm = SectionManager::instance();
284 const SectionInfo *si = sm.add(DString("missing_")+label,"requirements",1,section,SectionType::Section,1);
285 anchors.push_back(si);
286 doc += "\\htmlonly <div class=\"missing_";
287 doc += label;
288 doc += "\">\\endhtmlonly\n";
289 doc += DString("@section missing_")+label+" "+section+"\n"+text+"\n";
290 doc += "\\htmlonly </div>\\endhtmlonly\n";
291 }
292 };
293
294 // write list of requirements that do not have a satisfies relation
295 auto traceInfo = Config_getEnum(REQ_TRACEABILITY_INFO);
296 int numMissingSatisfied = static_cast<int>(missingSatisfiedRef.size());
297 if ((traceInfo==REQ_TRACEABILITY_INFO_t::YES || traceInfo==REQ_TRACEABILITY_INFO_t::UNSATISFIED_ONLY) && numMissingSatisfied>0)
298 {
299 writeMissingRef(missingSatisfiedRef,
300 "satisfies",
301 theTranslator->trUnsatisfiedRequirements(),
302 theTranslator->trUnsatisfiedRequirementsText(numMissingSatisfied==1,
303 writeMarkerList(theTranslator->trWriteList(numMissingSatisfied).str(),
304 numMissingSatisfied,
305 [&missingSatisfiedRef](size_t entryIndex) {
306 DString id = missingSatisfiedRef[entryIndex]->id();
307 return "@ref "+id + " \""+id+"\"";
308 })));
309 }
310
311 // write list of requirements that do not have a verifies relation
312 int numMissingVerified = static_cast<int>(missingVerifiedRef.size());
313 if ((traceInfo==REQ_TRACEABILITY_INFO_t::YES || traceInfo==REQ_TRACEABILITY_INFO_t::UNVERIFIED_ONLY) && numMissingVerified>0)
314 {
315 writeMissingRef(missingVerifiedRef,
316 "verifies",
317 theTranslator->trUnverifiedRequirements(),
318 theTranslator->trUnverifiedRequirementsText(numMissingVerified==1,
319 writeMarkerList(theTranslator->trWriteList(numMissingVerified).str(),
320 numMissingVerified,
321 [&missingVerifiedRef](size_t entryIndex) {
322 DString id = missingVerifiedRef[entryIndex]->id();
323 return "@ref "+id + " \""+id+"\"";
324 })));
325 }
326
327 AUTO_TRACE_ADD("doc=[[\n{}\n]]\n",doc);
328 p->reqPageDef->setDocumentation(doc,"requirements",1,false);
329 p->reqPageDef->addSectionsToDefinition(anchors);
330}

337void RequirementManager::addRequirementRefsForSymbol(const Definition *symbol)
338{
339 for (const auto &ref : symbol->requirementReferences())
340 {
341 Requirement *req = p->requirements.find(ref.reqId());
342 if (req)
343 {
344 //printf("adding reference from %s to requirement %s\n",qPrint(symbol->name()),qPrint(ref.reqId()));
345 switch (ref.type())
346 {
347 case RequirementRefType::Satisfies: req->addSatisfiedBy(symbol); break;
348 case RequirementRefType::Verifies: req->addVerifiedBy(symbol); break;
349 }
350 }
351 else
352 {
353 warn(ref.file(),ref.line(),"Reference to unknown requirement '{}' found",ref.reqId());
354 // invalid reference (file, and line needed)
355 }
356 }
357}

408void splitRequirementRefs(const RequirementRefs &inputReqRefs,RequirementRefs &satisfiesRefs,RequirementRefs &verifiesRefs)
409{
410 auto makeUnique = [](RequirementRefs &uniqueRefs)
411 {
412 // sort results on itemId
413 std::stable_sort(uniqueRefs.begin(),uniqueRefs.end(),
414 [](const auto &left,const auto &right)
415 { return left.reqId()< right.reqId() ||
416 (left.reqId()==right.reqId() &&
417 dstricmp(left.title(),right.title())<0);
418 });
419
420 // filter out duplicates
421 auto last = std::unique(uniqueRefs.begin(),uniqueRefs.end(),
422 [](const auto &left,const auto &right)
423 { return left.reqId()==right.reqId() &&
424 dstricmp(left.title(),right.title())==0;
425 });
426
427 // remove unused part
428 uniqueRefs.erase(last, uniqueRefs.end());
429 };
430
431 // split into satisfied and verifies references
432 std::partition_copy(
433 inputReqRefs.begin(),
434 inputReqRefs.end(),
435 std::back_inserter(satisfiesRefs),
436 std::back_inserter(verifiesRefs),
437 [](const auto &ref) { return ref.type()==RequirementRefType::Satisfies; }
438 );
439
440 // remove duplicates
441 makeUnique(satisfiesRefs);
442 makeUnique(verifiesRefs);
443}
