291StyleData::StyleData(const DString &reference, const DString &definition)
292{
293 reg::Match match;
294 if (reg::search(reference.str(),match,s_clause))
295 {
296 m_index = static_cast<int>(std::stoul(match[1].str()));
297 }
298 else // error
299 {
300 m_index = 0;
301 }
302 m_reference = reference;
303 m_definition = definition;

306bool StyleData::setStyle(const DString &command, const DString &styleName)
307{
308 reg::Match match;
309 if (!reg::search(command.str(),match,s_clause))
310 {
311 err("Style sheet '{}' contains no '\\s' clause.\n{{{}}}\n", styleName, command);
312 return false;
313 }
314 m_index = static_cast<int>(std::stoul(match[1].str()));
315
316 if (size_t index = command.find("\\sbasedon"); index!=DString::npos)
317 {
318 m_reference = command.left(index);
319 m_definition = command.mid(index);
320 }
321
322 return true;
323}

326void loadStylesheet(const DString &name, StyleDataMap& map)
327{
328 std::ifstream file(name.str());
329 if (!file.is_open())
330 {
331 err("Can't open RTF style sheet file {}. Using defaults.\n",name);
332 return;
333 }
334 msg("Loading RTF style sheet {}...\n",name);
335
336 uint32_t lineNr=1;
337
338 for (std::string line ; getline(file,line) ; ) // for each line
339 {
340 if (line.empty() || line[0]=='#') continue; // skip blanks & comments
341 static const reg::Ex assignment_splitter(R"(\s*=\s*)");
342 reg::Match match;
343 if (reg::search(line,match,assignment_splitter))
344 {
345 DString key = match.prefix().str();
346 DString value = match.suffix().str();
347 auto it = map.find(key.str());
348 if (it!=map.end())
349 {
350 StyleData& styleData = it->second;
351 styleData.setStyle(value,key);
352 }
353 else
354 {
355 warn(name,lineNr,"Unknown style sheet name {} ignored.",key);
356 }
357 }
358 else
359 {
360 warn(name,lineNr,"Assignment of style sheet name expected line='{}'!",line);
361 }
362 lineNr++;
363 }
364}
