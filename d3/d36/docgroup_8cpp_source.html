96int DocGroup::findExistingGroup(const MemberGroupInfo *info)
97{
98 std::lock_guard<std::mutex> lock(g_memberGroupInfoMapMutex);
99 //printf("findExistingGroup %s:%s\n",qPrint(info->header),qPrint(info->compoundName));
100 for (const auto &[groupId,groupInfo] : Doxygen::memberGroupInfoMap)
101 {
102 if (m_compoundName==groupInfo->compoundName && // same file or scope
103 !groupInfo->header.empty() && // not a nameless group
104 dstricmp(groupInfo->header,info->header)==0 // same header name
105 )
106 {
107 //printf("Found it!\n");
108 return groupId; // put the item in this group
109 }
110 }
111 return ++g_groupId; // start new group
112}

114void DocGroup::open(Entry *e,const DString &,int, bool implicit)
115{
116 if (!implicit) m_openCount++;
117 //printf("==> openGroup(name=%s,sec=%x) m_autoGroupStack=%zu\n",
118 // qPrint(e->name),e->section,m_autoGroupStack.size());
119 if (e->section.isGroupDoc()) // auto group
120 {
121 m_autoGroupStack.emplace_back(e->name,e->groupingPri());
122 }
123 else // start of a member group
124 {
125 //printf(" membergroup id=%d %s\n",m_memberGroupId,qPrint(m_memberGroupHeader));
126 if (m_memberGroupId==DOX_NOGROUP) // no group started yet
127 {
128 auto info = std::make_unique<MemberGroupInfo>();
129 info->header = m_memberGroupHeader.stripWhiteSpace();
130 info->compoundName = m_compoundName;
131 m_memberGroupId = findExistingGroup(info.get());
132 {
133 std::lock_guard<std::mutex> lock(g_memberGroupInfoMapMutex);
134 auto it = Doxygen::memberGroupInfoMap.find(m_memberGroupId);
135 if (it==Doxygen::memberGroupInfoMap.end())
136 {
137 //printf(" use membergroup %d\n",m_memberGroupId);
138 Doxygen::memberGroupInfoMap.emplace(m_memberGroupId,std::move(info));
139 }
140 }
141 m_memberGroupRelates = e->relates;
142 e->mGrpId = m_memberGroupId;
143 }
144 }
145}

147void DocGroup::close(Entry *e,const DString &fileName,int line,bool foundInline,bool implicit)
148{
149 if (!implicit)
150 {
151 if (m_openCount < 1)
152 {
153 warn(fileName,line,"unbalanced grouping commands");
154 }
155 else
156 {
157 m_openCount--;
158 }
159 }
160 //printf("==> closeGroup(name=%s,sec=%x,file=%s,line=%d) m_autoGroupStack=%zu\n",
161 // qPrint(e->name),e->section,qPrint(fileName),line,m_autoGroupStack.size());
162 if (m_memberGroupId!=DOX_NOGROUP) // end of member group
163 {
164 {
165 std::lock_guard<std::mutex> lock(g_memberGroupInfoMapMutex);
166 auto it = Doxygen::memberGroupInfoMap.find(m_memberGroupId);
167 if (it!=Doxygen::memberGroupInfoMap.end()) // known group
168 {
169 auto &info = it->second;
170 info->doc = m_memberGroupDocs;
171 //info->docFile = fileName;
172 //info->docLine = line;
173 }
174 }
175 m_memberGroupId=DOX_NOGROUP;
176 m_memberGroupRelates.clear();
177 m_memberGroupDocs.clear();
178 if (!foundInline)
179 {
180 e->mGrpId=DOX_NOGROUP;
181 e->relates="";
182 }
183 //printf("new group id=%d\n",m_memberGroupId);
184 }
185 else if (!m_autoGroupStack.empty()) // end of auto group
186 {
187 Grouping grp = m_autoGroupStack.back();
188 m_autoGroupStack.pop_back();
189 // see bug577005: we should not remove the last group for e
190 if (!foundInline && !e->groups.empty()) e->groups.pop_back();
191 //printf("Removing %s e=%p\n",qPrint(grp->groupname),e);
192 if (!foundInline) initGroupInfo(e);
193 }
194}

211void DocGroup::addDocs(Entry *e)
212{
213 if (e->section.isMemberGrp())
214 {
215 m_memberGroupDocs=e->brief.stripWhiteSpace();
216 e->doc = stripLeadingAndTrailingEmptyLines(e->doc,e->docLine);
217 if (!m_memberGroupDocs.empty() && !e->doc.empty())
218 {
219 m_memberGroupDocs+="\n\n";
220 }
221 m_memberGroupDocs+=e->doc;
222 {
223 std::lock_guard<std::mutex> lock(g_memberGroupInfoMapMutex);
224 auto it =Doxygen::memberGroupInfoMap.find(m_memberGroupId);
225 if (it!=Doxygen::memberGroupInfoMap.end())
226 {
227 auto &info = it->second;
228 info->doc = m_memberGroupDocs;
229 info->docFile = e->docFile;
230 info->docLine = e->docLine;
231 info->setRefItems(e->sli);
232 info->setRequirementReferences(e->rqli);
233 }
234 }
235 e->doc.clear();
236 e->brief.clear();
237 }
