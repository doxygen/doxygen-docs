285std::string Dir::filePath(const std::string &path,bool acceptsAbsPath) const
286{
287 std::string result;
288 if (acceptsAbsPath && !isRelativePath(path))
289 {
290 result = path;
291 }
292 else
293 {
294 result = fromFsPath(p->path / toFsPath(path));
295 }
296 correctPath(result);
297 return result;
298}

335bool Dir::copy(const std::string &srcName,const std::string &dstName,bool acceptsAbsPath) const
336{
337 const auto &copyOptions = fs::copy_options::overwrite_existing;
338 std::error_code ec, ec_perm;
339 std::string sn = filePath(srcName,acceptsAbsPath);
340 std::string dn = filePath(dstName,acceptsAbsPath);
341 fs::copy(toFsPath(sn),toFsPath(dn),copyOptions,ec);
342 // make sure the destination is writable for the owner (see issue #11600)
343 fs::permissions(toFsPath(dn), fs::perms::owner_write, fs::perm_options::add, ec_perm);
344 return !ec;
345}
