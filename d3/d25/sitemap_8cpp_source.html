66}

121void Crawlmap::finalize()
122{
123 std::sort(p->crawlLinks.begin(),p->crawlLinks.end());
124 p->crawlLinks.erase(std::unique(p->crawlLinks.begin(),p->crawlLinks.end()),p->crawlLinks.end());
125 for (auto &s : p->crawlLinks)
126 {
127 p->crawl << "<a href=\"" << s << "\"/>\n";
128 }
129
130 p->crawl << "</body>\n";
131 p->crawl << "</html>\n";
132 p->crawl.flush();
133 p->crawlFile.close();

143void Crawlmap::addContentsItem(bool, const DString &, const DString & ref,
144 const DString & file, const DString & anchor,
145 bool ,bool ,
146 const Definition *, const DString &)
147{
148 if (!file.empty() && ref.empty()) // made file optional param and
149 // don't place links in crawl file imported
150 // by tags
151 {
152 std::string link;
153 if (file[0]=='!' || file[0]=='^') // special markers for user defined URLs
154 {
155 link+=&file[1];
156 }
157 else
158 {
159 DString currFile = file;
160 addHtmlExtensionIfMissing(currFile);
161 DString currAnc = anchor;
162 link += currFile.data();
163 if (!currAnc.empty())
164 {
165 link += "#";
166 link += currAnc.str();
167 }
168 }
169 p->crawlLinks.push_back(link);
170 }
171}

173static DString makeFileName(const DString & withoutExtension)
174{
175 DString result=withoutExtension;
176 if (!result.empty())
177 {
178 if (result.at(0)=='!') // relative URL -> strip marker
179 {
180 result=result.mid(1);
181 }
182 else // add specified HTML extension
183 {
184 addHtmlExtensionIfMissing(result);
185 }
186 }
187 return result;
188}
189
190static DString makeRef(const DString & withoutExtension, const DString & anchor)
191{
192 if (withoutExtension.empty()) return DString();
193 DString result = makeFileName(withoutExtension);
194 if (anchor.empty()) return result;
195 return result+"#"+anchor;
196}

198void Crawlmap::addIndexItem(const Definition *context, const MemberDef *md,
199 const DString &sectionAnchor, const DString &title)
200{
201 if (context && md) // member
202 {
203 if (sectionAnchor.empty() && !md->hasDocumentation()) return;
204 DString cfname = md->getOutputFileBase();
205 DString anchor = !sectionAnchor.empty() ? sectionAnchor : md->anchor();
206 DString ref = makeRef(cfname, anchor);
207 p->crawlLinks.push_back(ref.str());
208 }
209 else if (context) // container
210 {
211 DString contRef = context->getOutputFileBase();
212 DString ref = makeRef(contRef,sectionAnchor);
213 p->crawlLinks.push_back(ref.str());
214 }
