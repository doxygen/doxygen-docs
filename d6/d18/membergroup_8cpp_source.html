44void MemberGroup::insertMember(MemberDef *md)
45{
46 //printf("MemberGroup::insertMember(%s) inSameSection=%d md->getSectionList()=%s\n",qPrint(md->name()),
47 // inSameSection,qPrint(md->getSectionList(m_container)->listType().to_string()));
48
49 MemberDef *firstMd = memberList->empty() ? nullptr : memberList->front();
50 if (inSameSection && firstMd &&
51 firstMd->getSectionList(m_container)!=md->getSectionList(m_container))
52 {
53 //printf("inSameSection=false\n");
54 inSameSection=false;
55 }
56 else if (inDeclSection==nullptr)
57 {
58 inDeclSection = const_cast<MemberList*>(md->getSectionList(m_container));
59 //printf("inDeclSection=%p type=%d\n",inDeclSection,inDeclSection->listType());
60 }
61 memberList->push_back(md);
62
63 // copy the group of the first member in the memberGroup
64 GroupDef *gd = nullptr;
65 if (firstMd && !firstMd->isAlias() && (gd=firstMd->getGroupDef()))
66 {
67 MemberDefMutable *mdm = toMemberDefMutable(md);
68 if (mdm)
69 {
70 mdm->setGroupDef(gd, firstMd->getGroupPri(),
71 firstMd->getGroupFileName(),
72 firstMd->getGroupStartLine(),
73 firstMd->getGroupHasDocs());
74 }
75 gd->insertMember(md);
76 }
77}

185const Definition *MemberGroup::memberContainer() const
186{
187 // return the container for the first member.
188 // Note this can be different from container() in case
189 // the member is rendered as part of a file but the members
190 // are actually of a namespace.
191 const Definition *ctx = nullptr;
192 if (memberList && !memberList->empty())
193 {
194 const MemberDef *md = memberList->front();
195 ctx = md->getClassDef();
196 if (ctx==nullptr) ctx = md->getNamespaceDef();
197 if (ctx==nullptr) ctx = md->getFileDef();
198 }
199 return ctx==nullptr ? m_container : ctx;
200}

208void MemberGroup::distributeMemberGroupDocumentation()
209{
210 //printf("MemberGroup::distributeMemberGroupDocumentation() %s\n",qPrint(grpHeader));
211 const MemberDef *md = nullptr;
212 for (const auto &smd : *memberList)
213 {
214 //printf("checking md=%s\n",qPrint(md->name()));
215 // find the first member of the group with documentation
216 if (!smd->documentation().empty() ||
217 !smd->briefDescription().empty() ||
218 !smd->inbodyDocumentation().empty()
219 )
220 {
221 //printf("found it!\n");
222 md = smd;
223 break;
224 }
225 }
226 if (md) // distribute docs of md to other members of the list
227 {
228 //printf("Member %s has documentation!\n",qPrint(md->name()));
229 for (const auto &iomd : *memberList)
230 {
231 MemberDefMutable *omd = toMemberDefMutable(iomd);
232 if (omd && md!=omd && omd->documentation().empty() &&
233 omd->briefDescription().empty() &&
234 omd->inbodyDocumentation().empty()
235 )
236 {
237 //printf("Copying documentation to member %s\n",qPrint(omd->name()));
238 omd->setBriefDescription(md->briefDescription(),md->briefFile(),md->briefLine());
239 omd->setDocumentation(md->documentation(),md->docFile(),md->docLine());
240 omd->setInbodyDocumentation(md->inbodyDocumentation(),md->inbodyFile(),md->inbodyLine());
241 }
242 }
243 }

312void addMembersToMemberGroup(MemberList *ml,
313 MemberGroupList *pMemberGroups,
314 const Definition *context)
315{
316 ASSERT(context!=nullptr);
317 //printf("addMemberToMemberGroup() context=%s\n",qPrint(context->name()));
318 if (ml==nullptr) return;
319
320 struct MoveMemberInfo
321 {
322 MoveMemberInfo(MemberDef *md,MemberGroup *mg,const RefItemVector &rv)
323 : memberDef(md), memberGroup(mg), sli(rv) {}
324 MemberDef *memberDef;
325 MemberGroup *memberGroup;
326 RefItemVector sli;
327 };
328 std::vector<MoveMemberInfo> movedMembers;
329
330 for (const auto &md : *ml)
331 {
332 if (md->isEnumerate()) // insert enum value of this enum into groups
333 {
334 for (const auto &fmd : md->enumFieldList())
335 {
336 int groupId=fmd->getMemberGroupId();
337 if (groupId!=-1)
338 {
339 auto it = Doxygen::memberGroupInfoMap.find(groupId);
340 if (it!=Doxygen::memberGroupInfoMap.end())
341 {
342 const auto &info = it->second;
343 auto mg_it = std::find_if(pMemberGroups->begin(),
344 pMemberGroups->end(),
345 [&groupId](const auto &g)
346 { return g->groupId()==groupId; }
347 );
348 MemberGroup *mg_ptr = nullptr;
349 if (mg_it==pMemberGroups->end())
350 {
351 auto mg = std::make_unique<MemberGroup>(
352 context,
353 groupId,
354 info->header,
355 info->doc,
356 info->docFile,
357 info->docLine,
358 ml->container());
359 mg_ptr = mg.get();
360 pMemberGroups->push_back(std::move(mg));
361 }
362 else
363 {
364 mg_ptr = (*mg_it).get();
365 }
366 mg_ptr->insertMember(fmd); // insert in member group
367 MemberDefMutable *fmdm = toMemberDefMutable(fmd);
368 if (fmdm)
369 {
370 fmdm->setMemberGroup(mg_ptr);
371 }
372 }
373 }
374 }
375 }
376 int groupId=md->getMemberGroupId();
377 if (groupId!=-1)
378 {
379 auto it = Doxygen::memberGroupInfoMap.find(groupId);
380 if (it!=Doxygen::memberGroupInfoMap.end())
381 {
382 const auto &info = it->second;
383 auto mg_it = std::find_if(pMemberGroups->begin(),
384 pMemberGroups->end(),
385 [&groupId](const auto &g)
386 { return g->groupId()==groupId; }
387 );
388 MemberGroup *mg_ptr = nullptr;
389 if (mg_it==pMemberGroups->end())
390 {
391 auto mg = std::make_unique<MemberGroup>(
392 context,
393 groupId,
394 info->header,
395 info->doc,
396 info->docFile,
397 info->docLine,
398 ml->container());
399 mg_ptr = mg.get();
400 pMemberGroups->push_back(std::move(mg));
401 }
402 else
403 {
404 mg_ptr = (*mg_it).get();
405 }
406 movedMembers.emplace_back(md,mg_ptr,info->m_sli);
407 }
408 }
409 }
410
411 // move the members to their group
412 for (const auto &mmi : movedMembers)
413 {
414 ml->remove(mmi.memberDef); // remove from member list
415 mmi.memberGroup->insertMember(mmi.memberDef->resolveAlias()); // insert in member group
416 mmi.memberGroup->setRefItems(mmi.sli);
417 MemberDefMutable *rmdm = toMemberDefMutable(mmi.memberDef);
418 if (rmdm)
419 {
420 rmdm->setMemberGroup(mmi.memberGroup);
421 }
422 }
423}
