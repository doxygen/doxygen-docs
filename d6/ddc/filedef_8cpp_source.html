52DString includeStatement(SrcLangExt lang,IncludeKind kind)
53{
54 bool isIDLorJava = lang==SrcLangExt::IDL || lang==SrcLangExt::Java;
55 if (isIDLorJava || (kind & IncludeKind_ImportMask))
56 {
57 return "import ";
58 }
59 else if (kind & IncludeKind_ObjCMask)
60 {
61 return "#import ";
62 }
63 else
64 {
65 return "#include ";
66 }
67}
68
69DString includeOpen(SrcLangExt lang,IncludeKind kind)
70{
71 if (lang==SrcLangExt::Java || kind==IncludeKind::ImportModule) return "";
72 if ((kind & IncludeKind_LocalMask) || (lang==SrcLangExt::IDL))
73 {
74 return "\"";
75 }
76 else
77 {
78 return "<";
79 }
80}
81
82DString includeClose(SrcLangExt lang,IncludeKind kind)
83{
84 if (lang==SrcLangExt::IDL) return "\";";
85 else if (lang==SrcLangExt::Java) return ";";
86
87 switch (kind)
88 {
89 case IncludeKind::ImportLocal: return "\";";
90 case IncludeKind::ImportLocalObjC: return "\"";
91 case IncludeKind::IncludeLocal: return "\"";
92 case IncludeKind::ImportSystem: return ">;";
93 case IncludeKind::ImportSystemObjC: return ">";
94 case IncludeKind::IncludeSystem: return ">";
95 case IncludeKind::ImportModule: return ";";
96 }
97 return "";
98}
99
100DString includeTagFileAttributes(SrcLangExt lang,IncludeKind kind)
101{
102 DString result;
103 bool isIDLorJava = lang==SrcLangExt::IDL || lang==SrcLangExt::Java;
104 result.sprintf("local=\"%s\" import=\"%s\" module=\"%s\" objc=\"%s\"",
105 (kind & IncludeKind_LocalMask) ? "yes" : "no",
106 (isIDLorJava || (kind & IncludeKind_ImportMask)) ? "yes" : "no",
107 (kind==IncludeKind::ImportModule) ? "yes" : "no",
108 (kind & IncludeKind_ObjCMask) ? "yes" : "no");
109 return result;
110}

374void FileDefImpl::writeTagFile(TextStream &tagFile)
375{
376 DString fn=getOutputFileBase();
377 addHtmlExtensionIfMissing(fn);
378 tagFile << " <compound kind=\"file\">\n";
379 tagFile << " <name>" << convertToXML(name()) << "</name>\n";
380 tagFile << " <path>" << convertToXML(stripFromPath(getPath())) << "</path>\n";
381 tagFile << " <filename>" << fn << "</filename>\n";
382 for (const auto &ii : m_includeList)
383 {
384 const FileDef *fd=ii.fileDef;
385 if (fd && fd->isLinkable() && !fd->isReference())
386 {
387 DString attr = includeTagFileAttributes(fd->getLanguage(),ii.kind);
388 tagFile << " <includes id=\""
389 << convertToXML(fd->getOutputFileBase()) << "\" "
390 << "name=\"" << convertToXML(fd->name()) << "\" "
391 << attr << ">"
392 << convertToXML(ii.includeName)
393 << "</includes>\n";
394 }
395 }
396 for (const auto &lde : LayoutDocManager::instance().docEntries(LayoutDocManager::File))
397 {
398 switch (lde->kind())
399 {
400 case LayoutDocEntry::FileClasses:
401 {
402 writeClassesToTagFile(tagFile, m_classes);
403 }
404 break;
405 case LayoutDocEntry::FileInterfaces:
406 {
407 writeClassesToTagFile(tagFile, m_interfaces);
408 }
409 break;
410 case LayoutDocEntry::FileStructs:
411 {
412 writeClassesToTagFile(tagFile, m_structs);
413 }
414 break;
415 case LayoutDocEntry::FileExceptions:
416 {
417 writeClassesToTagFile(tagFile, m_exceptions);
418 }
419 break;
420 case LayoutDocEntry::FileConcepts:
421 {
422 for (const auto *nd : m_concepts)
423 {
424 if (nd->isLinkableInProject())
425 {
426 tagFile << " <concept>" << convertToXML(nd->name()) << "</concept>\n";
427 }
428 }
429 }
430 break;
431 case LayoutDocEntry::FileNamespaces:
432 {
433 for (const auto *nd : m_namespaces)
434 {
435 if (nd->isLinkableInProject())
436 {
437 tagFile << " <namespace>" << convertToXML(nd->name()) << "</namespace>\n";
438 }
439 }
440 }
441 break;
442 case LayoutDocEntry::MemberDecl:
443 {
444 const LayoutDocEntryMemberDecl *lmd = dynamic_cast<const LayoutDocEntryMemberDecl*>(lde.get());
445 if (lmd)
446 {
447 MemberList * ml = getMemberList(lmd->type);
448 if (ml)
449 {
450 ml->writeTagFile(tagFile,false,false);
451 }
452 }
453 }
454 break;
455 case LayoutDocEntry::MemberGroups:
456 {
457 for (const auto &mg : m_memberGroups)
458 {
459 mg->writeTagFile(tagFile);
460 }
461 }
462 break;
463 default:
464 break;
465 }
466 }
467
468 writeDocAnchorsToTagFile(tagFile);
469 tagFile << " </compound>\n";
470}

472void FileDefImpl::writeDetailedDescription(OutputList &ol,const DString &title)
473{
474 if (hasDetailedDescription())
475 {
476 ol.pushGeneratorState();
477 ol.disable(OutputType::Html);
478 ol.writeRuler();
479 ol.popGeneratorState();
480 ol.pushGeneratorState();
481 ol.disableAllBut(OutputType::Html);
482 ol.writeAnchor(DString(),"details");
483 ol.popGeneratorState();
484 ol.startGroupHeader("details");
485 ol.parseText(title);
486 ol.endGroupHeader();
487
488 ol.startTextBlock();
489 if (!briefDescription().empty() && Config_getBool(REPEAT_BRIEF))
490 {
491 ol.generateDoc(briefFile(),
492 briefLine(),
493 this,
494 nullptr,
495 briefDescription(),
496 DocOptions());
497 }
498 if (!briefDescription().empty() && Config_getBool(REPEAT_BRIEF) &&
499 !documentation().empty())
500 {
501 ol.pushGeneratorState();
502 ol.disable(OutputType::Man);
503 ol.disable(OutputType::RTF);
504 ol.enableAll();
505 ol.disableAllBut(OutputType::Man);
506 ol.enable(OutputType::Latex);
507 ol.writeString("\n\n");
508 ol.popGeneratorState();
509 }
510 if (!documentation().empty())
511 {
512 ol.generateDoc(docFile(),
513 docLine(),
514 this,
515 nullptr,
516 documentation()+"\n",
517 DocOptions()
518 .setIndexWords(true));
519 }
520 //printf("Writing source ref for file %s\n",qPrint(name()));
521 if (Config_getBool(SOURCE_BROWSER))
522 {
523 ol.startParagraph("definition");
524 DString refText = theTranslator->trDefinedInSourceFile();
525 if (size_t fileMarkerPos = refText.find("@0"); fileMarkerPos!=DString::npos) // should always pass this.
526 {
527 ol.parseText(refText.left(fileMarkerPos)); //text left from marker 1
528 ol.writeObjectLink(DString(),getSourceFileBase(),DString(),name());
529 ol.parseText(refText.right(
530 refText.length()-fileMarkerPos-2)); // text right from marker 2
531 }
532 else
533 {
534 err("translation error: invalid marker in trDefinedInSourceFile()\n");
535 }
536 ol.endParagraph();
537 }
538 if (hasRequirementRefs()) writeRequirementRefs(ol);
539 ol.endTextBlock();
540 }

599void FileDefImpl::writeIncludeFiles(OutputList &ol)
600{
601 if (!m_includeList.empty())
602 {
603 ol.startTextBlock(true);
604 for (const auto &ii : m_includeList)
605 {
606 const FileDef *fd=ii.fileDef;
607 ol.startTypewriter();
608 SrcLangExt lang = fd ? fd->getLanguage() : SrcLangExt::Cpp;
609 ol.docify(::includeStatement(lang,ii.kind));
610 ol.docify(::includeOpen(lang,ii.kind));
611 ol.disable(OutputType::Html);
612 ol.docify(ii.includeName);
613 ol.enableAll();
614 ol.disableAllBut(OutputType::Html);
615
616 // Here we use the include file name as it appears in the file.
617 // we could also we the name as it is used within doxygen,
618 // then we should have used fd->docName() instead of ii->includeName
619 if (fd && fd->isLinkable())
620 {
621 ol.writeObjectLink(fd->getReference(),
622 fd->generateSourceFile() ? fd->includeName() : fd->getOutputFileBase(),
623 DString(),ii.includeName);
624 }
625 else
626 {
627 ModuleDef *mod = ModuleManager::instance().getPrimaryInterface(ii.includeName);
628 if (ii.kind==IncludeKind::ImportModule && mod && mod->isLinkable())
629 {
630 ol.writeObjectLink(mod->getReference(),mod->getOutputFileBase(),
631 DString(),ii.includeName);
632 }
633 else
634 {
635 ol.docify(ii.includeName);
636 }
637 }
638
639 ol.enableAll();
640 ol.docify(::includeClose(lang,ii.kind));
641 ol.endTypewriter();
642 ol.lineBreak();
643 }
644 ol.endTextBlock();
645 }
