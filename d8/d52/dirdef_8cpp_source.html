124DirDefImpl::DirDefImpl(const DString &path) : DefinitionMixin(path,1,1,path)
125{
126 bool fullPathNames = Config_getBool(FULL_PATH_NAMES);
127 // get display name (stripping the paths mentioned in STRIP_FROM_PATH)
128 // get short name (last part of path)
129 m_shortName = path;
130 m_diskName = path;
131 if (m_shortName.at(m_shortName.length()-1)=='/')
132 { // strip trailing /
133 m_shortName = m_shortName.left(m_shortName.length()-1);
134 }
135 if (size_t pi=m_shortName.rfind('/'); pi!=DString::npos)
136 { // remove everything till the last /
137 m_shortName = m_shortName.mid(pi+1);
138 }
139 setLocalName(m_shortName);
140 m_dispName = fullPathNames ? stripFromPath(path) : m_shortName;
141 if (m_dispName.length()>0 && m_dispName.at(m_dispName.length()-1)=='/')
142 { // strip trailing /
143 m_dispName = m_dispName.left(m_dispName.length()-1);
144 }
145
146 m_level=-1;
147 m_parent=nullptr;
148 m_hasDirectoryGraph=Config_getBool(DIRECTORY_GRAPH);
149
150}

701/** Add as "uses" dependency between \a this dir and \a dir,
702 * that was caused by a dependency on file \a fd.
703 * srcDirect and dstDirect indicate if it is a direct dependencies (true) or if
704 * the dependencies was indirect (e.g. a parent dir that has a child dir that has the dependencies)
705 */
706void DirDefImpl::addUsesDependency(const DirDef *dir,const FileDef *srcFd,
707 const FileDef *dstFd,bool srcDirect, bool dstDirect)
708{
709 if (this==dir) return; // do not add self-dependencies
710 AUTO_TRACE("add dependency {}->{} due to {}->{}",
711 qPrint(shortName()),
712 qPrint(dir->shortName()),
713 qPrint(srcFd->name()),
714 qPrint(dstFd->name()));
715
716 // levels match => add direct dependency
717 bool added=false;
718 UsedDir *usedDir = m_usedDirs.find(dir->getOutputFileBase());
719 if (usedDir) // dir dependency already present
720 {
721 const FilePair *usedPair = usedDir->findFilePair(FilePair::key(srcFd,dstFd));
722 if (usedPair==nullptr) // new file dependency
723 {
724 AUTO_TRACE_ADD("{} => {} new file dependency",srcFd->name(),dstFd->name());
725 usedDir->addFileDep(srcFd,dstFd, srcDirect, dstDirect);
726 added=true;
727 }
728 else
729 {
730 // dir & file dependency already added
731 }
732 }
733 else // new directory dependency
734 {
735 AUTO_TRACE_ADD("{} => {} new file dependency",srcFd->name(),dstFd->name());
736 auto newUsedDir = std::make_unique<UsedDir>(dir);
737 newUsedDir->addFileDep(srcFd,dstFd, srcDirect, dstDirect);
738 m_usedDirs.add(dir->getOutputFileBase(),std::move(newUsedDir));
739 added=true;
740 }
741 if (added)
742 {
743 if (dir->parent())
744 {
745 // add relation to parent of used dir
746 addUsesDependency(dir->parent(),
747 srcFd,
748 dstFd,
749 srcDirect,
750 false); // indirect dependency on dest dir
751 }
752 if (parent())
753 {
754 // add relation for the parent of this dir as well
755 parent()->addUsesDependency(dir,
756 srcFd,
757 dstFd,
758 false, // indirect dependency from source dir
759 dstDirect);
760 }
761 }
762}

783/** Computes the dependencies between directories
784 */
785void DirDefImpl::computeDependencies()
786{
787 AUTO_TRACE();
788 for (const auto &fd : m_fileList)
789 {
790 AUTO_TRACE_ADD("dir={} file={}",shortName(),fd->name());
791 for (const auto &ii : fd->includeFileList())
792 {
793 AUTO_TRACE_ADD("#include {}",ii.includeName);
794 if (ii.fileDef && ii.fileDef->isLinkable()) // linkable file
795 {
796 DirDef *usedDir = ii.fileDef->getDirDef();
797 if (usedDir)
798 {

1018/** In order to create stable, but unique directory names,
1019 * we compute the common part of the path shared by all directories.
1020 */
1021static void computeCommonDirPrefix()
1022{
1023 AUTO_TRACE();
1024 DString path;
1025 auto it = Doxygen::dirLinkedMap->begin();
1026 if (!Doxygen::dirLinkedMap->empty()) // we have at least one dir
1027 {
1028 // start will full path of first dir
1029 path=Portable::removeLongPathMarker((*it)->name());
1030 size_t i = path.length()>=2 ? path.rfind('/',path.length()-2) : DString::npos;
1031 bool done=false;
1032 if (i==DString::npos)
1033 {
1034 path="";
1035 }
1036 else
1037 {
1038 path=path.left(i+1);
1039 while (!done)
1040 {
1041 size_t l = path.length();
1042 size_t count=0;
1043 for (const auto &dir : *Doxygen::dirLinkedMap)
1044 {
1045 DString dirName = Portable::removeLongPathMarker(dir->name());
1046 //printf("dirName='%s' (l=%d) path='%s' (l=%d)\n",qPrint(dirName),dirName.length(),qPrint(path),path.length());
1047 if (dirName.length()>path.length())
1048 {
1049 if (dirName.left(l)!=path) // dirName does not start with path
1050 {
1051 i = l>=2 ? path.rfind('/',l-2) : DString::npos;
1052 if (i==DString::npos) // no unique prefix -> stop
1053 {
1054 path="";
1055 done=true;
1056 }
1057 else // restart with shorter path
1058 {
1059 path=path.left(i+1);
1060 break;
1061 }
1062 }
1063 }
1064 else // dir is shorter than path -> take path of dir as new start
1065 {
1066 path=dir->name();
1067 l = path.length();
1068 i = l>=2 ? path.rfind('/',l-2) : DString::npos;
1069 if (i==DString::npos) // no unique prefix -> stop
1070 {
1071 path="";
1072 done=true;
1073 }
1074 else // restart with shorter path
1075 {
1076 path=path.left(i+1);
1077 }
1078 break;
1079 }
1080 count++;
1081 }
1082 if (count==Doxygen::dirLinkedMap->size())
1083 // path matches for all directories -> found the common prefix
1084 {
1085 done=true;
1086 }
1087 }
1088 }
1089 }
1090 for (const auto &dir : *Doxygen::dirLinkedMap)
1091 {
1092 DString diskName = dir->name().right(dir->name().length()-path.length());
1093 dir->setDiskName(diskName);
