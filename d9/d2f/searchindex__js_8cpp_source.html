45void SearchTerm::makeTitle()
46{
47 if (std::holds_alternative<const Definition *>(info))
48 {
49 const Definition *def = std::get<const Definition *>(info);
50 Definition::DefType type = def->definitionType();
51 title = type==Definition::TypeGroup ? parseCommentAsHtml(def,nullptr,toGroupDef(def)->groupTitle(),def->getDefFileName(),def->getDefLine()) :
52 type==Definition::TypePage ? parseCommentAsHtml(def,nullptr,toPageDef(def)->title(),def->getDefFileName(),def->getDefLine()) :
53 def->localName();
54 }
55 else if (std::holds_alternative<const SectionInfo *>(info))
56 {
57 const SectionInfo *si = std::get<const SectionInfo *>(info);
58 title = parseCommentAsHtml(si->definition(),nullptr,si->title(),si->fileName(),si->lineNr());
59 }
60 else
61 {
62 ASSERT(false);
63 }
64}

66DString SearchTerm::termEncoded() const
67{
68 TextStream t;
69
70 for (size_t i=0;i<word.length();i++)
71 {
72 char c = word.at(i);
73 if (c<0 || isalnum(c))
74 {
75 t << word.at(i);
76 }
77 else // escape non-identifier characters
78 {
79 static const char *hex = "0123456789ABCDEF";
80 unsigned char uc = static_cast<unsigned char>(word.at(i));
81 t << '_';
82 t << hex[uc>>4];
83 t << hex[uc&0xF];
84 }
85 }
86
87 return convertUTF8ToLower(t.str());
88}

92//! helper function to simplify the given title string, and fill a list of start positions
93//! for the start of each word in the simplified title string.
94static void splitSearchTokens(DString &title,SizeVector &indices)
95{
96 if (title.empty()) return;
97
98 // simplify title to contain only words with single space as separator
99 size_t di=0;
100 bool lastIsSpace=true;
101 for (size_t si=0; si<title.length(); si++)
102 {
103 char c = title.at(si);
104 if (c=='@' || c=='\\') // skip over special commands
105 {
106 title.at(di)=' ';
107 if (si<title.length()-1)
108 {
109 c = title.at(++si);
110 while (si<title.length() && (isId(c) || c==':')) c = title.at(++si);
111 --si;
112 }
113 }
114 else if (c=='<') // skip over html tags
115 {
116 if (si<title.length()-1)
117 {
118 for (size_t tsi = si; tsi<title.length(); ++tsi)
119 {
120 if (title.at(tsi)=='>')
121 {
122 si=tsi;
123 break;
124 }
125 }
126 }
127 }
128 else if (isId(c) || c==':') // add "word" character
129 {
130 title.at(di)=c;
131 di++;
132 lastIsSpace=false;
133 }
134 else if (!lastIsSpace) // add one separator as space
135 {
136 title.at(di)=' ';
137 di++;
138 lastIsSpace=true;
139 }
140 }
141 if (di>0 && title.at(di-1)==' ') di--; // strip trailing whitespace
142 title.resize(di);
143
144 // create a list of start positions within title for
145 // each unique word in order of appearance
146 size_t p=0,i=0;
147 while ((i=title.find(' ',p))!=DString::npos)
148 {
149 std::string word = title.mid(p,i-p).str();
150 indices.push_back(p);
151 p = i+1;
152 }
153 if (p<title.length())
154 {
155 std::string word = title.mid(p).str();
156 indices.push_back(p);
157 }
158}

218static void addMemberToSearchIndex(const MemberDef *md)
219{
220 bool hideFriendCompounds = Config_getBool(HIDE_FRIEND_COMPOUNDS);
221 bool isLinkable = md->isLinkable();
222 const ClassDef *cd=nullptr;
223 const NamespaceDef *nd=nullptr;
224 const FileDef *fd=nullptr;
225 const GroupDef *gd=nullptr;
226 if (isLinkable &&
227 (
228 ((cd=md->getClassDef()) && cd->isLinkable() && !cd->isImplicitTemplateInstance()) ||
229 ((gd=md->getGroupDef()) && gd->isLinkable())
230 )
231 )
232 {
233 const DString &n = md->name();
234 if (!n.empty())
235 {
236 bool isFriendToHide = hideFriendCompounds &&
237 (md->typeString()=="friend class" ||
238 md->typeString()=="friend struct" ||
239 md->typeString()=="friend union");
240 if (!(md->isFriend() && isFriendToHide))
241 {
242 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,md));
243 }
244 if (md->isFunction() || md->isSlot() || md->isSignal())
245 {
246 g_searchIndexInfo[SEARCH_INDEX_FUNCTIONS].add(SearchTerm(n,md));
247 }
248 else if (md->isVariable())
249 {
250 g_searchIndexInfo[SEARCH_INDEX_VARIABLES].add(SearchTerm(n,md));
251 }
252 else if (md->isSequence())
253 {
254 g_searchIndexInfo[SEARCH_INDEX_SEQUENCES].add(SearchTerm(n,md));
255 }
256 else if (md->isDictionary())
257 {
258 g_searchIndexInfo[SEARCH_INDEX_DICTIONARIES].add(SearchTerm(n,md));
259 }
260 else if (md->isTypedef())
261 {
262 g_searchIndexInfo[SEARCH_INDEX_TYPEDEFS].add(SearchTerm(n,md));
263 }
264 else if (md->isEnumerate())
265 {
266 g_searchIndexInfo[SEARCH_INDEX_ENUMS].add(SearchTerm(n,md));
267 }
268 else if (md->isEnumValue())
269 {
270 g_searchIndexInfo[SEARCH_INDEX_ENUMVALUES].add(SearchTerm(n,md));
271 }
272 else if (md->isProperty())
273 {
274 g_searchIndexInfo[SEARCH_INDEX_PROPERTIES].add(SearchTerm(n,md));
275 }
276 else if (md->isEvent())
277 {
278 g_searchIndexInfo[SEARCH_INDEX_EVENTS].add(SearchTerm(n,md));
279 }
280 else if (md->isRelated() || md->isForeign() ||
281 (md->isFriend() && !isFriendToHide))
282 {
283 g_searchIndexInfo[SEARCH_INDEX_RELATED].add(SearchTerm(n,md));
284 }
285 }
286 }
287 else if (isLinkable &&
288 (((nd=md->getNamespaceDef()) && nd->isLinkable()) ||
289 ((fd=md->getFileDef()) && fd->isLinkable())
290 )
291 )
292 {
293 const DString &n = md->name();
294 if (!n.empty())
295 {
296 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,md));
297
298 if (md->isFunction())
299 {
300 g_searchIndexInfo[SEARCH_INDEX_FUNCTIONS].add(SearchTerm(n,md));
301 }
302 else if (md->isVariable())
303 {
304 g_searchIndexInfo[SEARCH_INDEX_VARIABLES].add(SearchTerm(n,md));
305 }
306 else if (md->isSequence())
307 {
308 g_searchIndexInfo[SEARCH_INDEX_SEQUENCES].add(SearchTerm(n,md));
309 }
310 else if (md->isDictionary())
311 {
312 g_searchIndexInfo[SEARCH_INDEX_DICTIONARIES].add(SearchTerm(n,md));
313 }
314 else if (md->isTypedef())
315 {
316 g_searchIndexInfo[SEARCH_INDEX_TYPEDEFS].add(SearchTerm(n,md));
317 }
318 else if (md->isEnumerate())
319 {
320 g_searchIndexInfo[SEARCH_INDEX_ENUMS].add(SearchTerm(n,md));
321 }
322 else if (md->isEnumValue())
323 {
324 g_searchIndexInfo[SEARCH_INDEX_ENUMVALUES].add(SearchTerm(n,md));
325 }
326 else if (md->isDefine())
327 {
328 g_searchIndexInfo[SEARCH_INDEX_DEFINES].add(SearchTerm(n,md));
329 }
330 }

336void createJavaScriptSearchIndex()
337{
338 // index classes
339 for (const auto &cd : *Doxygen::classLinkedMap)
340 {
341 if (cd->isLinkable())
342 {
343 DString n = cd->localName();
344 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,cd.get()));
345 if (Config_getBool(OPTIMIZE_OUTPUT_SLICE))
346 {
347 if (cd->compoundType()==ClassDef::Interface)
348 {
349 g_searchIndexInfo[SEARCH_INDEX_INTERFACES].add(SearchTerm(n,cd.get()));
350 }
351 else if (cd->compoundType()==ClassDef::Struct)
352 {
353 g_searchIndexInfo[SEARCH_INDEX_STRUCTS].add(SearchTerm(n,cd.get()));
354 }
355 else if (cd->compoundType()==ClassDef::Exception)
356 {
357 g_searchIndexInfo[SEARCH_INDEX_EXCEPTIONS].add(SearchTerm(n,cd.get()));
358 }
359 else // cd->compoundType()==ClassDef::Class
360 {
361 g_searchIndexInfo[SEARCH_INDEX_CLASSES].add(SearchTerm(n,cd.get()));
362 }
363 }
364 else // non slice optimization: group all types under classes
365 {
366 g_searchIndexInfo[SEARCH_INDEX_CLASSES].add(SearchTerm(n,cd.get()));
367 }
368 }
369 }
370
371 // index namespaces
372 for (const auto &nd : *Doxygen::namespaceLinkedMap)
373 {
374 if (nd->isLinkable())
375 {
376 DString n = nd->name();
377 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,nd.get()));
378 g_searchIndexInfo[SEARCH_INDEX_NAMESPACES].add(SearchTerm(n,nd.get()));
379 }
380 }
381
382 // index concepts
383 for (const auto &cd : *Doxygen::conceptLinkedMap)
384 {
385 if (cd->isLinkable())
386 {
387 DString n = cd->localName();
388 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,cd.get()));
389 g_searchIndexInfo[SEARCH_INDEX_CONCEPTS].add(SearchTerm(n,cd.get()));
390 }
391 }
392
393 // index modules
394 for (const auto &mod : ModuleManager::instance().modules())
395 {
396 if (mod->isLinkable() && mod->isPrimaryInterface())
397 {
398 DString n = mod->name();
399 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,mod.get()));
400 g_searchIndexInfo[SEARCH_INDEX_MODULES].add(SearchTerm(n,mod.get()));
401 }
402 }
403
404 // index files
405 for (const auto &fn : *Doxygen::inputNameLinkedMap)
406 {
407 for (const auto &fd : *fn)
408 {
409 DString n = fd->name();
410 if (fd->isLinkable())
411 {
412 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(n,fd.get()));
413 g_searchIndexInfo[SEARCH_INDEX_FILES].add(SearchTerm(n,fd.get()));
414 }
415 }
416 }
417
418 // index class members
419 {
420 // for each member name
421 for (const auto &mn : *Doxygen::memberNameLinkedMap)
422 {
423 // for each member definition
424 for (const auto &md : *mn)
425 {
426 addMemberToSearchIndex(md.get());
427 }
428 }
429 }
430
431 // index file/namespace members
432 {
433 // for each member name
434 for (const auto &mn : *Doxygen::functionNameLinkedMap)
435 {
436 // for each member definition
437 for (const auto &md : *mn)
438 {
439 addMemberToSearchIndex(md.get());
440 }
441 }
442 }
443
444 // index groups
445 for (const auto &gd : *Doxygen::groupLinkedMap)
446 {
447 if (gd->isLinkable())
448 {
449 DString title(filterTitle(gd->groupTitle()).str());
450 SizeVector tokenIndices;
451 splitSearchTokens(title,tokenIndices);
452 for (size_t index : tokenIndices)
453 {
454 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(title.mid(index),gd.get()));
455 g_searchIndexInfo[SEARCH_INDEX_GROUPS].add(SearchTerm(title.mid(index),gd.get()));
456 }
457 }
458 }
459
460 // index pages
461 for (const auto &pd : *Doxygen::pageLinkedMap)
462 {
463 if (pd->isLinkable())
464 {
465 DString title(filterTitle(pd->title()).str());
466 SizeVector tokenIndices;
467 splitSearchTokens(title,tokenIndices);
468 for (size_t index : tokenIndices)
469 {
470 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(title.mid(index),pd.get()));
471 g_searchIndexInfo[SEARCH_INDEX_PAGES].add(SearchTerm(title.mid(index),pd.get()));
472 }
473 }
474 }
475
476 // main page
477 if (Doxygen::mainPage)
478 {
479 DString title(filterTitle(Doxygen::mainPage->title()).str());
480 SizeVector tokenIndices;
481 splitSearchTokens(title,tokenIndices);
482 for (size_t index : tokenIndices)
483 {
484 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(title.mid(index),Doxygen::mainPage.get()));
485 g_searchIndexInfo[SEARCH_INDEX_PAGES].add(SearchTerm(title.mid(index),Doxygen::mainPage.get()));
486 }
487 }
488
489 // sections
490 const auto &sm = SectionManager::instance();
491 for (const auto &sectionInfo : sm)
492 {
493 if (sectionInfo->level()>0) // level 0 is for page titles
494 {
495 DString title = filterTitle(sectionInfo->title());
496 SizeVector tokenIndices;
497 splitSearchTokens(title,tokenIndices);
498 //printf("split(%s)=(%s) %zu\n",qPrint(sectionInfo->title()),qPrint(title),tokenIndices.size());
499 for (size_t index : tokenIndices)
500 {
501 g_searchIndexInfo[SEARCH_INDEX_ALL].add(SearchTerm(title.mid(index),sectionInfo.get()));
502 g_searchIndexInfo[SEARCH_INDEX_PAGES].add(SearchTerm(title.mid(index),sectionInfo.get()));
503 }
504 }
505 }
506
507 // sort all lists
508 for (auto &sii : g_searchIndexInfo) // for each index
509 {
510 for (auto &[name,symList] : sii.symbolMap) // for each symbol in the index
511 {
512 // sort the symbols (first on search term, and then on full name)
513 //
514 // `std::stable_sort` is used here due to reproducibility issues
515 // on key collisions
516 // https://github.com/doxygen/doxygen/issues/10445
517 std::stable_sort(symList.begin(),
518 symList.end(),
519 [](const auto &t1,const auto &t2)
520 {
521 int eq = dstricmp_sort(t1.word,t2.word); // search term first
522 return eq==0 ? dstricmp_sort(t1.title,t2.title)<0 : eq<0; // then full title
523 });
524 }
525 }
526}

528static void writeJavascriptSearchData(const DString &searchDirName)
529{
530 std::ofstream t = Portable::openOutputStream(searchDirName+"/searchdata.js");
531 if (t.is_open())
532 {
533 t << "var indexSectionsWithContent =\n";
534 t << "{\n";
535 int j=0;
536 for (const auto &sii : g_searchIndexInfo)
537 {
538 if (!sii.symbolMap.empty())
539 {
540 if (j>0) t << ",\n";
541 t << " " << j << ": \"";
542
543 std::string previous_letter; // start with value that does not exist in the map
544 for (const auto &[letter,list] : sii.symbolMap)
545 {
546 if (letter != previous_letter)
547 {
548 if ( letter == "\"" ) t << "\\"; // add escape for backslash
549 t << letter;
550 previous_letter = letter;
551 }
552 }
553 t << "\"";
554 j++;
555 }
556 }
557 if (j>0) t << "\n";
558 t << "};\n\n";
559 t << "var indexSectionNames =\n";
560 t << "{\n";
561 j=0;
562 for (const auto &sii : g_searchIndexInfo)
563 {
564 if (!sii.symbolMap.empty())
565 {
566 if (j>0) t << ",\n";
567 t << " " << j << ": \"" << convertToJSString(sii.name,true,false) << "\"";
568 j++;
569 }
570 }
571 if (j>0) t << "\n";
572 t << "};\n\n";
573 t << "var indexSectionLabels =\n";
574 t << "{\n";
575 j=0;
576 for (const auto &sii : g_searchIndexInfo)
577 {
578 if (!sii.symbolMap.empty())
579 {
580 if (j>0) t << ",\n";
581 t << " " << j << ": \"" << convertToJSString(convertToXML(sii.getText()),true,false) << "\"";
582 j++;
583 }
584 }
585 if (j>0) t << "\n";
586 t << "};\n\n";
587 }

590static void writeJavasScriptSearchDataPage(const DString &baseName,const DString &dataFileName,const SearchIndexList &list)
591{
592 auto isDef = [](const SearchTerm::LinkInfo &info)
593 {
594 return std::holds_alternative<const Definition *>(info);
595 };
596 auto getDef = [&isDef](const SearchTerm::LinkInfo &info)
597 {
598 return isDef(info) ? std::get<const Definition *>(info) : nullptr;
599 };
600 auto isSection = [](const SearchTerm::LinkInfo &info)
601 {
602 return std::holds_alternative<const SectionInfo *>(info);
603 };
604 auto getSection = [&isSection](const SearchTerm::LinkInfo &info)
605 {
606 return isSection(info) ? std::get<const SectionInfo *>(info) : nullptr;
607 };
608
609 int cnt = 0;
610 std::ofstream ti = Portable::openOutputStream(dataFileName);
611 if (!ti.is_open())
612 {
613 err("Failed to open file '{}' for writing...\n",dataFileName);
614 return;
615 }
616
617 ti << "var searchData=\n";
618 // format
619 // searchData[] = array of items
620 // searchData[x][0] = id
621 // searchData[x][1] = [ name + child1 + child2 + .. ]
622 // searchData[x][1][0] = name as shown
623 // searchData[x][1][y+1] = info for child y
624 // searchData[x][1][y+1][0] = url
625 // searchData[x][1][y+1][1] = 1 => target="_parent"
626 // searchData[x][1][y+1][1] = 0 => target="_blank"
627 // searchData[x][1][y+1][2] = scope
628
629 ti << "[\n";
630 bool firstEntry=true;
631
632 int childCount=0;
633 DString lastWord;
634 const Definition *prevScope = nullptr;
635 for (auto it = list.begin(); it!=list.end();)
636 {
637 const SearchTerm &term = *it;
638 const SearchTerm::LinkInfo info = term.info;
639 const Definition *d = getDef(info);
640 const SectionInfo *si = getSection(info);
641 ASSERT(d || si); // either d or si should be valid
642 DString word = term.word;
643 DString id = term.termEncoded();
644 ++it;
645 const Definition *scope = d ? d->getOuterScope() : nullptr;
646 const SearchTerm::LinkInfo next = it!=list.end() ? it->info : SearchTerm::LinkInfo();
647 const Definition *nextScope = isDef(next) ? getDef(next)->getOuterScope() : nullptr;
648 const MemberDef *md = toMemberDef(d);
649 DString anchor = d ? d->anchor() : si ? si->label() : DString();
650
651 if (word!=lastWord) // this item has a different search word
652 {
653 if (!firstEntry)
654 {
655 ti << "]]]";
656 ti << ",\n";
657 }
658 firstEntry=false;
659 ti << " ['" << id << "_" << cnt++ << "',['";
660 if (next==SearchTerm::LinkInfo() || it->word!=word) // unique result, show title
661 {
662 ti << convertToJSString(convertToXML(term.title),true,true);
663 }
664 else // multiple results, show matching word only, expanded list will show title
665 {
666 ti << convertToJSString(convertToXML(term.word),true,true);
667 }
668 ti << "',[";
669 childCount=0;
670 prevScope=nullptr;
671 }
672
673 if (childCount>0)
674 {
675 ti << "],[";
676 }
677 DString fn = d ? d->getOutputFileBase() : si ? si->fileName() : DString();
678 DString ref = d ? d->getReference() : si ? si->ref() : DString();
679 addHtmlExtensionIfMissing(fn);
680 DString extRef = externalRef("../",ref)+fn;
681 if (!anchor.empty())
682 {
683 extRef+="#"+anchor;
684 }
685 ti << "'" << convertToJSString(extRef,true,true) << "',";
686
687 bool extLinksInWindow = Config_getBool(EXT_LINKS_IN_WINDOW);
688 if (!extLinksInWindow || ref.empty())
689 {
690 ti << "1,";
691 }
692 else
693 {
694 ti << "0,";
695 }
696
697 if (lastWord!=word && (next==SearchTerm::LinkInfo() || it->word!=word)) // unique search result
698 {
699 if (d && d->getOuterScope()!=Doxygen::globalScope)
700 {
701 ti << "'" << convertToJSString(convertToXML(d->getOuterScope()->name()),true,true) << "'";
702 }
703 else if (md)
704 {
705 const FileDef *fd = md->getBodyDef();
706 if (fd==nullptr) fd = md->getFileDef();
707 if (fd)
708 {
709 ti << "'" << convertToJSString(convertToXML(fd->localName()),true,true) << "'";
710 }
711 }
712 else
713 {
714 ti << "''";
715 }
716 }
717 else // multiple entries with the same name
718 {
719 bool found=false;
720 bool overloadedFunction = ((prevScope!=nullptr && scope==prevScope) || (scope && scope==nextScope)) &&
721 md && md->isCallable();
722 DString prefix;
723 if (md) prefix=convertToXML(md->localName());
724 if (overloadedFunction) // overloaded member function
725 {
726 prefix+=convertToXML(md->argsString());
727 // show argument list to disambiguate overloaded functions
728 }
729 else if (md && md->isCallable()) // unique member function
730 {
731 prefix+="()"; // only to show it is a callable symbol
732 }
733 DString name;
734 if (d)
735 {
736 switch (d->definitionType())
737 {
738 case Definition::TypeClass: name = convertToXML((toClassDef(d))->displayName()); found=true; break;
739 case Definition::TypeNamespace: name = convertToXML((toNamespaceDef(d))->displayName()); found=true; break;
740 case Definition::TypeModule: name = convertToXML(d->name()+" "+theTranslator->trModule(false,true)); found=true; break;
741 case Definition::TypePage: name = convertToXML(filterTitle(toPageDef(d)->title())); found=true; break;
742 case Definition::TypeGroup: name = convertToXML(filterTitle(toGroupDef(d)->groupTitle())); found=true; break;
743 default:
744 if (scope==nullptr || scope==Doxygen::globalScope) // in global scope
745 {
746 if (md)
747 {
748 const FileDef *fd = md->getBodyDef();
749 if (fd==nullptr) fd = md->resolveAlias()->getFileDef();
750 if (fd)
751 {
752 if (!prefix.empty()) prefix+=":&#160;";
753 name = prefix + convertToXML(fd->localName());
754 found = true;
755 }
756 }
757 }
758 else if (md && (md->resolveAlias()->getClassDef() || md->resolveAlias()->getNamespaceDef()))
759 // member in class or namespace scope
760 {
761 SrcLangExt lang = md->getLanguage();
762 DString sep = getLanguageSpecificSeparator(lang);
763 name = convertToXML(d->getOuterScope()->qualifiedName()) + sep + prefix;
764 found = true;
765 }
766 else if (scope) // some thing else? -> show scope
767 {
768 name = prefix + convertToXML(scope->name());
769 found = true;
770 }
771 break;
772 }
773 }
774 else if (si)
775 {
776 name = parseCommentAsHtml(si->definition(),nullptr,si->title(),si->fileName(),si->lineNr());
777 name = convertToXML(name);
778 found = true;
779 }
780 if (!found) // fallback
781 {
782 name = prefix + "("+theTranslator->trGlobalNamespace()+")";
783 }
784
785 ti << "'" << convertToJSString(name,true,true) << "'";
786
787 prevScope = scope;
788 childCount++;
789 }
790 lastWord = word;
791 }
792 if (!firstEntry)
793 {
794 ti << "]]]\n";
795 }
796 ti << "];\n";
797 Doxygen::indexList->addStyleSheetFile(("search/"+baseName+".js").data());
798}

801void writeJavaScriptSearchIndex()
802{
803 // write index files
804 DString searchDirName = Config_getString(HTML_OUTPUT)+"/search";
805
806 std::size_t numThreads = static_cast<std::size_t>(Config_getInt(NUM_PROC_THREADS));
807 if (numThreads>1) // multi threaded version
808 {
809 ThreadPool threadPool(numThreads);
810 std::vector< std::future<int> > results;
811 for (auto &sii : g_searchIndexInfo)
812 {
813 int p=0;
814 for (const auto &[letter,symList] : sii.symbolMap)
815 {
816 DString baseName;
817 baseName.sprintf("%s_%x",sii.name.data(),p);
818 DString dataFileName = searchDirName + "/"+baseName+".js";
819 auto &list = symList;
820 auto processFile = [p,baseName,dataFileName,&list]()
821 {
822 writeJavasScriptSearchDataPage(baseName,dataFileName,list);
823 return p;
824 };
825 results.emplace_back(threadPool.queue(processFile));
826 p++;
827 }
828 }
829 // wait for the results
830 for (auto &f : results) f.get();
831 }
832 else // single threaded version
833 {
834 for (auto &sii : g_searchIndexInfo)
835 {
836 int p=0;
837 for (const auto &[letter,symList] : sii.symbolMap)
838 {
839 DString baseName;
840 baseName.sprintf("%s_%x",sii.name.data(),p);
841 DString dataFileName = searchDirName + "/"+baseName+".js";
842 writeJavasScriptSearchDataPage(baseName,dataFileName,symList);
843 p++;
844 }
845 }
846 }
847
848 writeJavascriptSearchData(searchDirName);
849 auto &mgr = ResourceMgr::instance();
850 {
851 std::ofstream fn = Portable::openOutputStream(searchDirName+"/search.js");
852 if (fn.is_open())
853 {
854 TextStream t(&fn);
855 t << substitute(mgr.getAsString("search.js"),"$PROJECTID",getProjectId());
856 }
857 }
858
859 Doxygen::indexList->addStyleSheetFile("search/searchdata.js");
860 Doxygen::indexList->addStyleSheetFile("search/search.js");
861}
