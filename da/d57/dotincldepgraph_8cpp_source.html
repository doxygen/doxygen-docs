26void DotInclDepGraph::buildGraph(DotNode *n,const FileDef *fd,int distance)
27{
28 const IncludeInfoList &includeFiles = m_inverse ? fd->includedByFileList() : fd->includeFileList();
29 for (const auto &ii : includeFiles)
30 {
31 const FileDef *bfd = ii.fileDef;
32 DString in = ii.includeName;
33 //printf(">>>> in='%s' bfd=%p\n",qPrint(ii->includeName),bfd);
34 bool doc=true,src=false;
35 if (bfd)
36 {
37 in = bfd->absFilePath();
38 doc = bfd->isLinkable() && !bfd->isHidden();
39 src = bfd->generateSourceFile();
40 }
41 if (doc || src || !Config_getBool(HIDE_UNDOC_RELATIONS))
42 {
43 DString url="";
44 if (bfd) url=bfd->getOutputFileBase();
45 if (!doc && src)
46 {
47 url=bfd->getSourceFileBase();
48 }
49 auto it = m_usedNodes.find(in.str());
50 if (it!=m_usedNodes.end()) // file is already a node in the graph
51 {
52 DotNode *bn = it->second;
53 n->addChild(bn,EdgeInfo::Blue,EdgeInfo::Solid);
54 bn->addParent(n);
55 bn->setDistance(distance);
56 }
57 else
58 {
59 DString tmp_url;
60 DString tooltip;
61 if (bfd)
62 {
63 tmp_url=doc || src ? bfd->getReference()+"$"+url : DString();
64 tooltip = bfd->briefDescriptionAsTooltip();
65 }
66 DotNode *bn = new DotNode(this,
67 ii.includeName, // label
68 tooltip, // tip
69 tmp_url, // url
70 false, // rootNode
71 nullptr); // cd
72 n->addChild(bn,EdgeInfo::Blue,EdgeInfo::Solid);
73 bn->addParent(n);
74 m_usedNodes.emplace(in.str(),bn);
75 bn->setDistance(distance);
76
77 if (bfd) buildGraph(bn,bfd,distance+1);
78 }
79 }
80 }
81}

83void DotInclDepGraph::determineVisibleNodes(DotNodeDeque &queue, int &maxNodes)
84{
85 while (!queue.empty() && maxNodes>0)
86 {
87 DotNode *n = queue.front();
88 queue.pop_front();
89 if (!n->isVisible() && n->distance()<=Config_getInt(MAX_DOT_GRAPH_DEPTH)) // not yet processed
90 {
91 n->markAsVisible();
92 maxNodes--;
93 // add direct children
94 for (const auto &dn : n->children())
95 {
96 queue.push_back(dn);
97 }
98 }
99 }
100}

102void DotInclDepGraph::determineTruncatedNodes(DotNodeDeque &queue)
103{
104 while (!queue.empty())
105 {
106 DotNode *n = queue.front();
107 queue.pop_front();
108 if (n->isVisible() && n->isTruncated()==DotNode::Unknown)
109 {
110 bool truncated = false;
111 for (const auto &dn : n->children())
112 {
113 if (!dn->isVisible())
114 {
115 truncated = true;
116 }
117 else
118 {
119 queue.push_back(dn);
120 }
121 }
122 n->markAsTruncated(truncated);
123 }
124 }
125}

127DotInclDepGraph::DotInclDepGraph(const FileDef *fd,bool inverse)
128{
129 m_inverse = inverse;
130 ASSERT(fd!=nullptr);
131 m_inclDepFileName = fd->includeDependencyGraphFileName();
132 m_inclByDepFileName = fd->includedByDependencyGraphFileName();
133 DString tmp_url=fd->getReference()+"$"+fd->getOutputFileBase();
134 DString tooltip = fd->briefDescriptionAsTooltip();
135 m_startNode = new DotNode(this,
136 fd->docName(),
137 tooltip,
138 tmp_url,
139 true); // root node
140 m_startNode->setDistance(0);
141 m_usedNodes.emplace(fd->absFilePath().str(),m_startNode);
142 buildGraph(m_startNode,fd,1);
143
144 int maxNodes = Config_getInt(DOT_GRAPH_MAX_NODES);
145 DotNodeDeque openNodeQueue;
146 openNodeQueue.push_back(m_startNode);
147 determineVisibleNodes(openNodeQueue,maxNodes);
148 openNodeQueue.clear();
149 openNodeQueue.push_back(m_startNode);
150 determineTruncatedNodes(openNodeQueue);
151}
