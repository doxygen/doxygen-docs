157/** Class representing a list of different code generators.
158 * It offers the same interface as the specific code generators,
159 * and will forward each method to all generators that are enabled.
160 */
161class OutputCodeList
162{
163 public:
164 using OutputCodeIntfPtr = std::unique_ptr<OutputCodeIntf>;
165
166 private:
167 struct OutputCodeElem
168 {
169 explicit OutputCodeElem(OutputCodeIntfPtr &&p) : intf(std::move(p)) {}
170 OutputCodeElem(const OutputCodeElem &other)
171 {
172 intf = other.intf->clone();
173 enabled = other.enabled;
174 }
175 OutputCodeElem &operator=(const OutputCodeElem &other)
176 {
177 if (&other!=this)
178 {
179 intf = other.intf->clone();
180 enabled = other.enabled;
181 }
182 return *this;
183 }
184 OutputCodeIntfPtr intf;
185 bool enabled = true;
186 };
187 public:
188
189 int id() const { return m_id; }
190 void setId(int id) { m_id = id; }
191
192 void add(OutputCodeIntfPtr &&p)
193 {
194 m_outputCodeList.emplace_back(std::move(p));
195 }
196
197 /** Add a code generator to the list, using a syntax similar to std::make_unique<T>() */
198 template<class T,class... As>
199 T* add(As&&... args)
200 {
201 add(std::make_unique<T>(std::forward<As>(args)...));
202 return static_cast<T*>(m_outputCodeList.back().intf.get());
203 }
204
205 /** Returns a pointer to a specific generator in the list */
206 template<class T>
207 T *get(OutputType o)
208 {
209 for (auto &e : m_outputCodeList)
210 {
211 if (e.intf->type()==o) { return static_cast<T*>(e.intf.get()); }
212 }
213 return nullptr;
214 }
215
216 /** Enable or disable a specific generator */
217 void setEnabledFiltered(OutputType o,bool enabled)
218 {
219 for (auto &e : m_outputCodeList)
220 {
221 if (e.intf->type()==o) e.enabled = enabled;
222 }
223 }
224
225 void clear()
226 {
227 m_outputCodeList.clear();
228 }

286 { foreach(&OutputCodeIntf::endFold); }
287
288 private:
289 template<class... Ts, class... As>
290 void foreach(void (OutputCodeIntf::*methodPtr)(Ts...),As&&... args)
291 {
292 for (auto &e : m_outputCodeList)
293 {
294 if (e.enabled)
295 {
296 (e.intf.get()->*methodPtr)(std::forward<As>(args)...);
297 }
298 }
299 }

301 std::vector<OutputCodeElem> m_outputCodeList;
302 int m_id = -1;
