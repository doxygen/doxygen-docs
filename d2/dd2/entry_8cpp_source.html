172void Entry::removeSubEntry(const Entry *e)
173{
174 auto it = std::find_if(m_sublist.begin(),m_sublist.end(),
175 [e](const std::shared_ptr<Entry>&elem) { return elem.get()==e; });
176 if (it!=m_sublist.end())
177 {
178 m_sublist.erase(it);
179 }
180}
