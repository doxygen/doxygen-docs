29/** @brief Base class representing a piece of a documentation page */
30struct LayoutDocEntry
31{
32 ABSTRACT_BASE_CLASS(LayoutDocEntry)
33
34#define ENTRY_SPECIFICATIONS \
35 /* Generic items for all pages */ \
36 ESPEC(MemberGroups) \
37 ESPEC(MemberDeclStart) ESPEC(MemberDeclEnd) ESPEC(MemberDecl) \
38 ESPEC(MemberDefStart) ESPEC(MemberDefEnd) ESPEC(MemberDef) \
39 ESPEC(BriefDesc) ESPEC(DetailedDesc) \
40 ESPEC(AuthorSection) ESPEC(Includes) \
41 /* Class specific items */ \
42 ESPEC(ClassInlineClasses) \
43 ESPEC(ClassInheritanceGraph) ESPEC(ClassNestedClasses) \
44 ESPEC(ClassCollaborationGraph) ESPEC(ClassAllMembersLink) \
45 ESPEC(ClassUsedFiles) \
46 /* Concept specific items */ \
47 ESPEC(ConceptDefinition) \
48 /* Namespace specific items */ \
49 ESPEC(NamespaceNestedNamespaces) ESPEC(NamespaceNestedConstantGroups) \
50 ESPEC(NamespaceClasses) ESPEC(NamespaceConcepts) ESPEC(NamespaceInterfaces) ESPEC(NamespaceStructs) \
51 ESPEC(NamespaceRecords) ESPEC(NamespaceExceptions) ESPEC(NamespaceInlineClasses) \
52 /* File specific items */ \
53 ESPEC(FileClasses) ESPEC(FileConcepts) ESPEC(FileInterfaces) ESPEC(FileStructs) ESPEC(FileRecords) \

62 ESPEC(GroupDirs) ESPEC(GroupNestedGroups) ESPEC(GroupFiles) \

64 /* Directory specific items */ \
65 ESPEC(DirSubDirs) ESPEC(DirFiles) ESPEC(DirGraph)
66
67 enum Kind {
68#define ESPEC(x) x,
69 ENTRY_SPECIFICATIONS
70#undef ESPEC
71 };
72 virtual Kind kind() const = 0;
73 std::string entryToString() const
74 {
75 switch (kind())
76 {
77#define ESPEC(x) case x: return #x; break;
78 ENTRY_SPECIFICATIONS
79#undef ESPEC
80 default: return "unknown"; // to satisfy compiler
81 }
82 }
83 virtual std::string id() const = 0;
84 virtual bool visible() const = 0;
85};
