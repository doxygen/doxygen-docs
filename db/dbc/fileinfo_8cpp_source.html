106std::string FileInfo::absFilePath() const
107{
108 std::string result;
109 std::error_code ec;
110 fs::path path = toFsPath(m_name);
111 if (!path.is_relative())
112 {
113 result = fromFsPath(path.lexically_normal());
114 }
115 else
116 {
117 result = fromFsPath((fs::current_path(ec) / path).lexically_normal());
118 }
119 correctPath(result);
120 return result;
121}

142std::string FileInfo::dirPath(bool absPath) const
143{
144 std::string result;
145 if (absPath)
146 {
147 result = absFilePath();
148 }
149 else
150 {
151 result = m_name;
152 correctPath(result);
153 }
154 size_t pos = result.rfind('/');
155 if (pos==std::string::npos)
156 {
157 return ".";
158 }
159 else if (pos==0)
160 {
161 return "/";
162 }
163 else
164 {
165 return result.substr(0,pos);
166 }
167}
