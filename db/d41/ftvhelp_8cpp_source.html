78int FTVNode::computeTreeDepth(int level) const
79{
80 int maxDepth=level;
81 for (const auto &n : children)
82 {
83 if (!n->children.empty())
84 {
85 int d = n->computeTreeDepth(level+1);
86 if (d>maxDepth) maxDepth=d;
87 }
88 }
89 return maxDepth;
90}
91
92int FTVNode::numNodesAtLevel(int level,int maxLevel) const
93{
94 int num=0;
95 if (level<maxLevel)
96 {
97 num++; // this node
98 for (const auto &n : children)
99 {
100 num+=n->numNodesAtLevel(level+1,maxLevel);
101 }
102 }
103 return num;
104}

142/*! Increase the level of the contents hierarchy.

146void FTVHelp::incContentsDepth()

153/*! Decrease the level of the contents hierarchy.
154 * This will end the current sublist.
155 * \sa incContentsDepth()
156 */
157void FTVHelp::decContentsDepth()
158{
159 //printf("%p: decContentsDepth() indent=%d\n",this,p->indent);
160 ASSERT(p->indent>0);
161 if (p->indent>0)
162 {
163 p->indent--;
164 auto &nl = p->indentNodes[p->indent];
165 if (!nl.empty())
166 {
167 auto &parent = nl.back();
168 auto &children = p->indentNodes[p->indent+1];
169 for (const auto &child : children)
170 {
171 parent->children.push_back(child);
172 }
173 children.clear();
174 }
175 }
176}

219static DString node2URL(const FTVNodePtr &n,bool overruleFile=false,bool srcLink=false)
220{
221 DString url = n->file;
222 if (!url.empty() && url.at(0)=='!') // relative URL
223 {
224 // remove leading !
225 url = url.mid(1);
226 }
227 else if (!url.empty() && url.at(0)=='^') // absolute URL
228 {
229 // skip, keep ^ in the output
230 }
231 else // local file (with optional anchor)
232 {
233 if (overruleFile && n->def && n->def->definitionType()==Definition::TypeFile)
234 {
235 const FileDef *fd = toFileDef(n->def);
236 if (srcLink)
237 {
238 url = fd->getSourceFileBase();
239 }
240 else
241 {
242 url = fd->getOutputFileBase();
243 }
244 }
245 addHtmlExtensionIfMissing(url);
246 if (!n->anchor.empty()) url+="#"+n->anchor;
247 }
248 return url;
249}

263static void generateIndent(TextStream &t, const FTVNodePtr &n,bool opened)
264{
265 bool dynamicSections = Config_getBool(HTML_DYNAMIC_SECTIONS);
266 int indent=0;
267 auto parent = n->parent.lock();
268 while (parent) { indent++; parent=parent->parent.lock(); }
269 if (n->isDir && dynamicSections)
270 {
271 const char *ARROW_DOWN = "<span class=\"arrowhead opened\"></span>";
272 const char *ARROW_RIGHT = "<span class=\"arrowhead closed\"></span>";
273 DString dir = opened ? ARROW_DOWN : ARROW_RIGHT;
274 for(int i=0;i<indent;i++) t << "<span class=\"spacer\">&#160;</span>";
275 t << "<span id=\"arr_" << generateIndentLabel(n,0) << "\" class=\"arrow\">" << dir
276 << "</span>";
277 }
278 else
279 {
280 for(int i=0;i<=indent;i++) t << "<span class=\"spacer\">&#160;</span>";
281 }

284void FTVHelp::Private::generateLink(TextStream &t,const FTVNodePtr &n)
285{
286 //printf("FTVHelp::generateLink(ref=%s,file=%s,anchor=%s\n",
287 // qPrint(n->ref),qPrint(n->file),qPrint(n->anchor));
288 bool setTarget = false;
289 bool nameAsHtml = !n->nameAsHtml.empty();
290 DString text = nameAsHtml ? n->nameAsHtml : convertToHtml(n->name);
291 if (n->file.empty()) // no link
292 {
293 t << "<b>" << text << "</b>";
294 }
295 else // link into other frame
296 {
297 if (!n->ref.empty()) // link to entity imported via tag file
298 {
299 t << "<a class=\"elRef\" ";
300 DString result = externalLinkTarget();
301 if (result != "") setTarget = true;
302 t << result;
303 }
304 else // local link
305 {
306 t << "<a class=\"el\" ";
307 }
308 t << "href=\"";
309 t << externalRef("",n->ref);
310 t << node2URL(n);
311 if (!setTarget)
312 {
313 if (topLevelIndex)
314 t << "\" target=\"basefrm\">";
315 else
316 t << "\" target=\"_self\">";
317 }
318 else
319 {
320 t << "\">";
321 }
322 t << text;
323 t << "</a>";
324 if (!n->ref.empty())
325 {
326 t << "&#160;[external]";
327 }
328 }
329}

331static void generateBriefDoc(TextStream &t,const Definition *def)
332{
333 DString brief = def->briefDescription(true);
334 //printf("*** %p: generateBriefDoc(%s)='%s'\n",def,qPrint(def->name()),qPrint(brief));
335 if (!brief.empty())
336 {
337 auto parser { createDocParser() };
338 auto ast { validatingParseDoc(*parser.get(),
339 def->briefFile(),
340 def->briefLine(),
341 def,
342 nullptr,
343 brief,
344 DocOptions()
345 .setSingleLine(true)
346 .setLinkFromIndex(true))
347 };
348 const DocNodeAST *astImpl = dynamic_cast<const DocNodeAST*>(ast.get());
349 if (astImpl)
350 {
351 DString relPath = relativePathToRoot(def->getOutputFileBase());
352 OutputCodeList htmlList;
353 htmlList.add<HtmlCodeGenerator>(&t,relPath);
354 HtmlDocVisitor visitor(t,htmlList,def);
355 std::visit(visitor,astImpl->root);
356 }
357 }

381void FTVHelp::Private::generateTree(TextStream &t, const FTVNodes &nl,int level,int maxLevel,int &index)
382{
383 bool dynamicSections = Config_getBool(HTML_DYNAMIC_SECTIONS);
384 for (const auto &n : nl)
385 {
386 t << "<tr id=\"row_" << generateIndentLabel(n,0) << "\"";
387 t << " class=\"";
388 if ((index&1)==0) // even row
389 t << "even";
390 else
391 t << "odd";
392 if (level>=maxLevel && dynamicSections) // item invisible by default
393 t << " hidden";
394 else // item visible by default
395 index++;
396 t << "\"";
397 t << "><td class=\"entry\">";
398 bool nodeOpened = level+1<maxLevel;
399 generateIndent(t,n,nodeOpened);
400 if (n->isDir)
401 {
402 if (n->def && n->def->definitionType()==Definition::TypeGroup)
403 {
404 // no icon
405 }
406 else if (n->def && n->def->definitionType()==Definition::TypePage)
407 {
408 // no icon
409 }
410 else if (n->def && n->def->definitionType()==Definition::TypeNamespace)
411 {
412 if ((n->def->getLanguage() == SrcLangExt::Slice) || (n->def->getLanguage() == SrcLangExt::Fortran))
413 {
414 t << "<span class=\"icona\"><span class=\"icon\">M</span></span>";
415 }
416 else if ((n->def->getLanguage() == SrcLangExt::Java) || (n->def->getLanguage() == SrcLangExt::VHDL))
417 {
418 t << "<span class=\"icona\"><span class=\"icon\">P</span></span>";
419 }
420 else
421 {
422 t << "<span class=\"icona\"><span class=\"icon\">N</span></span>";
423 }
424 }
425 else if (n->def && n->def->definitionType()==Definition::TypeModule)
426 {
427 t << "<span class=\"icona\"><span class=\"icon\">M</span></span>";
428 }
429 else if (n->def && n->def->definitionType()==Definition::TypeClass)
430 {
431 char icon=compoundIcon(toClassDef(n->def));
432 t << "<span class=\"icona\"><span class=\"icon\">" << icon << "</span></span>";
433 }
434 else if (dynamicSections)
435 {
436 t << "<span id=\"img_" << generateIndentLabel(n,0) << "\" class=\"iconfolder\">"
437 << "<div class=\"folder-icon"
438 << (nodeOpened ? " open" : "")
439 << "\"></div></span>";
440 }
441 generateLink(t,n);
442 t << "</td><td class=\"desc\">";
443 if (n->def)
444 {
445 generateBriefDoc(t,n->def);
446 }
447 t << "</td></tr>\n";
448 folderId++;
449 generateTree(t,n->children,level+1,maxLevel,index);
450 }
451 else // leaf node
452 {
453 const FileDef *srcRef=nullptr;
454 if (n->def && n->def->definitionType()==Definition::TypeFile &&
455 (toFileDef(n->def))->generateSourceFile())
456 {
457 srcRef = toFileDef(n->def);
458 }
459 if (srcRef)
460 {
461 DString fn=srcRef->getSourceFileBase();
462 addHtmlExtensionIfMissing(fn);
463 t << "<a href=\"" << fn << "\">";
464 }
465 if (n->def && n->def->definitionType()==Definition::TypeGroup)
466 {
467 // no icon
468 }
469 else if (n->def && n->def->definitionType()==Definition::TypePage)
470 {
471 // no icon
472 }
473 else if (n->def && n->def->definitionType()==Definition::TypeNamespace)
474 {
475 if ((n->def->getLanguage() == SrcLangExt::Slice) || (n->def->getLanguage() == SrcLangExt::Fortran))
476 {
477 t << "<span class=\"icona\"><span class=\"icon\">M</span></span>";
478 }
479 else if ((n->def->getLanguage() == SrcLangExt::Java) || (n->def->getLanguage() == SrcLangExt::VHDL))
480 {
481 t << "<span class=\"icona\"><span class=\"icon\">P</span></span>";
482 }
483 else
484 {
485 t << "<span class=\"icona\"><span class=\"icon\">N</span></span>";
486 }
487 }
488 else if (n->def && n->def->definitionType()==Definition::TypeModule)
489 {
490 t << "<span class=\"icona\"><span class=\"icon\">M</span></span>";
491 }
492 else if (n->def && n->def->definitionType()==Definition::TypeClass)
493 {
494 char icon=compoundIcon(toClassDef(n->def));
495 t << "<span class=\"icona\"><span class=\"icon\">" << icon << "</span></span>";
496 }
497 else if (n->def && n->def->definitionType()==Definition::TypeConcept)
498 {
499 t << "<span class=\"icona\"><span class=\"icon\">R</span></span>";
500 }
501 else if (n->def && n->def->definitionType()==Definition::TypeDir)
502 {
503 t << "<span class=\"iconfolder\"><div class=\"folder-icon\"></div></span>";
504 }
505 else
506 {
507 t << "<span class=\"icondoc\"><div class=\"doc-icon\"></div></span>";
508 }
509 if (srcRef)
510 {
511 t << "</a>";
512 }
513 generateLink(t,n);
514 t << "</td><td class=\"desc\">";
515 if (n->def)
516 {
517 generateBriefDoc(t,n->def);
518 }
519 t << "</td></tr>\n";
520 }
521 }

576static DString convertFileId2Var(const DString &fileId)
577{
578 DString varId = fileId;
579 size_t i=varId.rfind('/');
580 if (i!=DString::npos) varId = varId.mid(i+1);
581 if (isdigit(varId[0])) varId.prepend("_");
582
583 return substitute(varId,"-","_");
584}

596static void collectJSTreeFiles(const FTVNodes &nl,JSTreeFiles &files)
597{

617static std::mutex g_navIndexMutex;
618
619static bool generateJSTree(NavIndexEntryList &navIndex,TextStream &t,
620 const FTVNodes &nl,int level,bool &first)
621{
622 DString htmlOutput = Config_getString(HTML_OUTPUT);
623 DString indentStr;
624 indentStr.fill(' ',level*2);
625
626 bool found=false;
627 for (const auto &n : nl)
628 {
629 // terminate previous entry
630 if (!first) t << ",\n";
631 first=false;
632
633 // start entry
634 if (!found)
635 {
636 t << "[\n";
637 }
638 found=true;
639
640 if (n->addToNavIndex) // add entry to the navigation index
641 {
642 std::lock_guard lock(g_navIndexMutex);
643 if (n->def && n->def->definitionType()==Definition::TypeFile)
644 {
645 const FileDef *fd = toFileDef(n->def);
646 bool src = false;
647 bool doc = fd->visibleInIndex(src);
648 if (doc)
649 {
650 navIndex.emplace_back(node2URL(n,true,false),pathToNode(n,n));
651 }
652 if (src)
653 {
654 navIndex.emplace_back(node2URL(n,true,true),pathToNode(n,n));
655 }
656 }
657 else
658 {
659 navIndex.emplace_back(node2URL(n),pathToNode(n,n));
660 }
661 }
662
663 if (n->separateIndex) // store items in a separate file for dynamic loading
664 {
665 t << indentStr << " [ ";
666 generateJSLink(t,n);
667 if (!n->children.empty()) // write children to separate file for dynamic loading
668 {
669 DString fileId = n->file;
670 if (!n->anchor.empty())
671 {
672 fileId+="_"+n->anchor;
673 }
674 if (dupOfParent(n))
675 {
676 fileId+="_dup";
677 }
678 t << "\"" << fileId << "\" ]";
679 }
680 else // no children
681 {
682 t << "null ]";
683 }
684 }
685 else // show items in this file
686 {
687 bool firstChild=true;
688 t << indentStr << " [ ";
689 generateJSLink(t,n);
690 bool emptySection = !generateJSTree(navIndex,t,n->children,level+1,firstChild);
691 if (emptySection)
692 t << "null ]";
693 else
694 t << "\n" << indentStr << " ] ]";
695 }
696 }
697 return found;
698}

700static void generateJSTreeFiles(NavIndexEntryList &navIndex,TextStream &t,const FTVNodes &nodeList)
701{
702 DString htmlOutput = Config_getString(HTML_OUTPUT);
703
704 auto getVarName = [](const FTVNodePtr n)
705 {
706 DString fileId = n->file;
707 if (!n->anchor.empty()) fileId+="_"+n->anchor;
708 if (dupOfParent(n)) fileId+="_dup";
709 return fileId;
710 };
711
712 auto generateJSFile = [&](const JSTreeFile &tf)
713 {
714 DString fileId = getVarName(tf.node);
715 DString fileName = htmlOutput+"/"+fileId+".js";
716 std::ofstream ff = Portable::openOutputStream(fileName);
717 if (ff.is_open())
718 {
719 bool firstChild = true;
720 TextStream tt(&ff);
721 tt << "var " << convertFileId2Var(fileId) << " =\n";
722 generateJSTree(navIndex,tt,tf.node->children,1,firstChild);
723 tt << "\n];";
724 }
725 };
726
727 JSTreeFiles jsTreeFiles;
728 collectJSTreeFiles(nodeList,jsTreeFiles);
729
730 std::size_t numThreads = static_cast<std::size_t>(Config_getInt(NUM_PROC_THREADS));
731 if (numThreads>1) // multi threaded version
732 {
733 ThreadPool threadPool(numThreads);
734 std::vector< std::future<void> > results;
735 for (const auto &tf : jsTreeFiles)
736 {
737 results.emplace_back(threadPool.queue([&](){ generateJSFile(tf); }));
738 }
739 // wait for the results
740 for (auto &f : results) f.get();
741 }
742 else // single threaded version
743 {
744 for (const auto &tf : jsTreeFiles)
745 {
746 generateJSFile(tf);
747 }
748 }
749}

834 tsidx << "\n";
835

837 if (it!=navIndex.end() && elemCount>=maxElemCount) // switch to new sub-index
838 {
839 tsidx << "};\n";
840 elemCount=0;
841 tsidx.close();
842 subIndex++;
843 DString fileName = htmlOutput+"/navtreeindex"+DString().setNum(subIndex)+".js";
844 tsidx = Portable::openOutputStream(fileName);
845 if (!tsidx.is_open()) break;
846 tsidx << "var NAVTREEINDEX" << subIndex << " =\n";
847 tsidx << "{\n";
848 }
849 }
850 tsidx << "};\n";
851 t << "\n];\n";
852 }
853 t << "\nconst SYNCONMSG = '" << theTranslator->trPanelSynchronisationTooltip(false) << "';";
854 t << "\nconst SYNCOFFMSG = '" << theTranslator->trPanelSynchronisationTooltip(true) << "';";
855 t << "\nconst LISTOFALLMEMBERS = '" << theTranslator->trListOfAllMembers() << "';";
856 }
857
858 auto &mgr = ResourceMgr::instance();
859 {

861 if (fn.is_open())
862 {
863 TextStream t(&fn);
864 t << substitute(
865 substitute(mgr.getAsString("navtree.js"),
866 "$TREEVIEW_WIDTH", DString().setNum(Config_getInt(TREEVIEW_WIDTH))),
867 "$PROJECTID",getProjectId());

884void FTVHelp::generateTreeViewInline(TextStream &t)
885{
886 bool dynamicSections = Config_getBool(HTML_DYNAMIC_SECTIONS);
887 int preferredNumEntries = Config_getInt(HTML_INDEX_NUM_ENTRIES);
888 t << "<div class=\"directory\">\n";
889 int d=1, depth=1;
890 for (const auto &n : p->indentNodes[0])
891 {
892 if (!n->children.empty())
893 {
894 d = n->computeTreeDepth(2);
895 if (d>depth) depth=d;
896 }
897 }
898 int preferredDepth = depth;
899 // write level selector
900 if (depth>1)
901 {
902 if (dynamicSections)
903 {
904 t << "<div class=\"levels\">[";
905 t << theTranslator->trDetailLevel();
906 t << " ";
907 for (int i=1;i<=depth;i++)
908 {
909 t << "<span class=\"dyn-level-" << i << "\">" << i << "</span>";
910 }
911 t << "]</div>";
912 }
913
914 if (preferredNumEntries>0)
915 {
916 preferredDepth=1;
917 for (int i=1;i<=depth;i++)
918 {
919 int num=0;
920 for (const auto &n : p->indentNodes[0])
921 {
922 num+=n->numNodesAtLevel(0,i);
923 }
924 if (num<=preferredNumEntries)
925 {
926 preferredDepth=i;
927 }
928 else
929 {
930 break;
931 }
932 }
933 }
934 }
935 //printf("preferred depth=%d\n",preferredDepth);
936
937 if (!p->indentNodes[0].empty())
938 {
939 t << "<table class=\"directory\">\n";
940 int index=0;
941 p->generateTree(t,p->indentNodes[0],0,preferredDepth,index);
942 t << "</table>\n";
943 }
944
945 t << "</div><!-- directory -->\n";
946}
