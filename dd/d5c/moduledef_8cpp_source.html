276void ModuleDefImpl::addMemberToModule(const Entry *root,MemberDef *md)
277{
278 AUTO_TRACE("{}:{} member {} of module {} exported={}",
279 qPrint(root->fileName),root->startLine,
280 qPrint(md->qualifiedName()),qPrint(name()),
281 root->exported);
282 MemberList *allMemberList = getMemberList(MemberListType::AllMembersList());
283 if (allMemberList==nullptr)
284 {
285 m_memberLists.emplace_back(std::make_unique<MemberList>(MemberListType::AllMembersList(),MemberListContainer::Module));
286 allMemberList = m_memberLists.back().get();
287 }
288 if (allMemberList->contains(md))
289 {
290 return;
291 }
292 allMemberList->push_back(md);
293 switch (md->memberType())
294 {
295 case MemberType::Variable:
296 addMemberToList(MemberListType::DecVarMembers(),md);
297 break;
298 case MemberType::Function:
299 addMemberToList(MemberListType::DecFuncMembers(),md);
300 break;
301 case MemberType::Typedef:
302 addMemberToList(MemberListType::DecTypedefMembers(),md);
303 break;
304 case MemberType::Enumeration:
305 addMemberToList(MemberListType::DecEnumMembers(),md);
306 break;
307 default:
308 break;
309 }
310 auto mdm = toMemberDefMutable(md);
311 if (mdm && root->exported && !md->isExported())
312 {
313 mdm->setExported(true);
314 }

317void ModuleDefImpl::mergeSymbolsFrom(ModuleDefImpl *other)
318{
319 AUTO_TRACE("{} merging symbols of {} ({}:{})",
320 name(),other->qualifiedName(),other->getDefFileName(),other->getDefLine());
321 for (const auto &cd : other->getClasses())
322 {
323 m_classes.add(cd->qualifiedName(),cd);
324 }
325 for (const auto &cd : other->getConcepts())
326 {
327 m_concepts.add(cd->qualifiedName(),cd);
328 }
329 auto mergeMemberList = [this,other](MemberListType lt)
330 {
331 const auto &srcMl = other->getMemberList(lt);
332 if (srcMl)
333 {
334 auto &dstMl = m_memberLists.get(lt,srcMl->container());
335 for (const auto &md : *srcMl)
336 {
337 dstMl->push_back(md);
338 }
339 }
340 };
341 mergeMemberList(MemberListType::DecVarMembers());
342 mergeMemberList(MemberListType::DecFuncMembers());
343 mergeMemberList(MemberListType::DecTypedefMembers());
344 mergeMemberList(MemberListType::DecEnumMembers());
345}

347void ModuleDefImpl::writeDocumentation(OutputList &ol)
348{
349 if (isReference()) return;
350 bool generateTreeView = Config_getBool(GENERATE_TREEVIEW);
351 ol.pushGeneratorState();
352 AUTO_TRACE("%s file=%s",name(),getDefFileName());
353 SrcLangExt lang = getLanguage();
354 DString pageTitle;
355 if (Config_getBool(HIDE_COMPOUND_REFERENCE))
356 {
357 pageTitle = displayName();
358 }
359 else
360 {
361 pageTitle = theTranslator->trModuleReference(displayName());
362 }
363 startFile(ol,getOutputFileBase(),false,name(),pageTitle,HighlightedItem::ModuleVisible,false,DString());
364
365 // ---- title part
366 ol.startHeaderSection();
367 bool writeOutlinePanel = generateTreeView && Config_getBool(PAGE_OUTLINE_PANEL);
368 if (!writeOutlinePanel) writeSummaryLinks(ol);
369 ol.startTitleHead(getOutputFileBase());
370
371 ol.pushGeneratorState();
372 ol.disable(OutputType::Man);
373 ol.parseText(pageTitle);
374 ol.popGeneratorState();
375
376 addGroupListToTitle(ol,this);
377
378 ol.pushGeneratorState();
379 ol.disable(OutputType::Man);
380 ol.endTitleHead(getOutputFileBase(),pageTitle);
381 ol.popGeneratorState();
382
383 ol.pushGeneratorState();
384 ol.disableAllBut(OutputType::Man);
385 ol.endTitleHead(getOutputFileBase(),name());
386 ol.writeString(" - ");
387 ol.parseText(pageTitle);
388 ol.popGeneratorState();
389
390 ol.endHeaderSection();
391 ol.startContents();
392
393 //---------------------------------------- start flexible part -------------------------------
394 for (const auto &lde : LayoutDocManager::instance().docEntries(LayoutDocManager::Module))
395 {
396 const LayoutDocEntrySection *ls = dynamic_cast<const LayoutDocEntrySection*>(lde.get());
397 switch (lde->kind())
398 {
399 case LayoutDocEntry::BriefDesc:
400 writeBriefDescription(ol);
401 break;
402 case LayoutDocEntry::MemberDeclStart:
403 startMemberDeclarations(ol);
404 break;
405 case LayoutDocEntry::ModuleClasses:
406 if (ls) writeClassDeclarations(ol,ls->title(lang));
407 break;
408 case LayoutDocEntry::ModuleConcepts:
409 if (ls) writeConcepts(ol,ls->title(lang));
410 break;
411 case LayoutDocEntry::ModuleExports:
412 if (ls) writeExports(ol,ls->title(lang));
413 break;
414 case LayoutDocEntry::ModuleUsedFiles:
415 if (ls) writeFiles(ol,ls->title(lang));
416 break;
417 case LayoutDocEntry::MemberGroups:
418 writeMemberGroups(ol);
419 break;
420 case LayoutDocEntry::MemberDecl:
421 {
422 const LayoutDocEntryMemberDecl *lmd = dynamic_cast<const LayoutDocEntryMemberDecl*>(lde.get());
423 if (lmd) writeMemberDeclarations(ol,lmd->type,lmd->title(lang));
424 }
425 break;
426 case LayoutDocEntry::MemberDeclEnd:
427 endMemberDeclarations(ol);
428 break;
429 case LayoutDocEntry::DetailedDesc:
430 if (ls) writeDetailedDescription(ol,ls->title(lang));
431 break;
432 case LayoutDocEntry::MemberDefStart:
433 startMemberDocumentation(ol);
434 break;
435 case LayoutDocEntry::MemberDef:
436 {
437 const LayoutDocEntryMemberDef *lmd = dynamic_cast<const LayoutDocEntryMemberDef*>(lde.get());
438 if (lmd) writeMemberDocumentation(ol,lmd->type,lmd->title(lang));
439 }
440 break;
441 case LayoutDocEntry::MemberDefEnd:
442 endMemberDocumentation(ol);
443 break;
444 case LayoutDocEntry::AuthorSection:
445 writeAuthorSection(ol);
446 break;
447 case LayoutDocEntry::ClassIncludes:
448 case LayoutDocEntry::ClassInheritanceGraph:
449 case LayoutDocEntry::ClassNestedClasses:
450 case LayoutDocEntry::ClassCollaborationGraph:
451 case LayoutDocEntry::ClassAllMembersLink:
452 case LayoutDocEntry::ClassUsedFiles:
453 case LayoutDocEntry::ClassInlineClasses:
454 case LayoutDocEntry::FileClasses:
455 case LayoutDocEntry::FileConcepts:
456 case LayoutDocEntry::FileInterfaces:
457 case LayoutDocEntry::FileStructs:
458 case LayoutDocEntry::FileExceptions:
459 case LayoutDocEntry::FileNamespaces:
460 case LayoutDocEntry::FileConstantGroups:
461 case LayoutDocEntry::FileIncludes:
462 case LayoutDocEntry::FileIncludeGraph:
463 case LayoutDocEntry::FileIncludedByGraph:
464 case LayoutDocEntry::FileInlineClasses:
465 case LayoutDocEntry::FileSourceLink:
466 case LayoutDocEntry::NamespaceNestedNamespaces:
467 case LayoutDocEntry::NamespaceNestedConstantGroups:
468 case LayoutDocEntry::NamespaceClasses:
469 case LayoutDocEntry::NamespaceConcepts:
470 case LayoutDocEntry::NamespaceInterfaces:
471 case LayoutDocEntry::NamespaceStructs:
472 case LayoutDocEntry::NamespaceExceptions:
473 case LayoutDocEntry::NamespaceInlineClasses:
474 case LayoutDocEntry::ConceptDefinition:
475 case LayoutDocEntry::GroupClasses:
476 case LayoutDocEntry::GroupConcepts:
477 case LayoutDocEntry::GroupModules:
478 case LayoutDocEntry::GroupInlineClasses:
479 case LayoutDocEntry::GroupNamespaces:
480 case LayoutDocEntry::GroupDirs:
481 case LayoutDocEntry::GroupNestedGroups:
482 case LayoutDocEntry::GroupFiles:
483 case LayoutDocEntry::GroupGraph:
484 case LayoutDocEntry::GroupPageDocs:
485 case LayoutDocEntry::DirSubDirs:
486 case LayoutDocEntry::DirFiles:
487 case LayoutDocEntry::DirGraph:
488 err("Internal inconsistency: member '{}' should not be part of "
489 "LayoutDocManager::Module entry list\n",lde->entryToString());
490 break;
491 }
492 }
493
494 //---------------------------------------- end flexible part -------------------------------
495 if (generateTreeView)
496 {
497 ol.pushGeneratorState();
498 ol.disableAllBut(OutputType::Html);
499 ol.endContents();
500 ol.writeString("</div><!-- doc-content -->\n");
501 writePageNavigation(ol);
502 ol.writeString("</div><!-- container -->\n");
503 ol.popGeneratorState();
504 }
505 endFile(ol,generateTreeView,true);
506
507 ol.popGeneratorState();
508}

580 ol.enableAll();
581 ol.disableAllBut(OutputType::Man);
582 ol.enable(OutputType::Latex);

584 ol.popGeneratorState();
585 }
586 if (!documentation().empty())
587 {
588 ol.generateDoc(docFile(),
589 docLine(),
590 this,
591 nullptr,
592 documentation()+"\n",
593 DocOptions()
594 .setIndexWords(true));
595 }

734void ModuleDefImpl::addMembersToMemberGroup()
735{
736 for (auto &ml : m_memberLists)
737 {
738 if (ml->listType().isDeclaration())
739 {
740 ::addMembersToMemberGroup(ml.get(),&m_memberGroups,this);
741 }
742 }
743
744 // add members inside sections to their groups
745 for (const auto &mg : m_memberGroups)
746 {
747 if (mg->allMembersInSameSection() && Config_getBool(SUBGROUPING))

781void ModuleDefImpl::sortMemberLists()
782{
783 for (auto &ml : m_memberLists)
784 {
785 if (ml->needsSorting()) { ml->sort(); ml->setNeedsSorting(false); }
786 }
787
788 if (Config_getBool(SORT_BRIEF_DOCS))
789 {
790 auto classComp = [](const ClassLinkedRefMap::Ptr &c1,const ClassLinkedRefMap::Ptr &c2)
791 {
792 return Config_getBool(SORT_BY_SCOPE_NAME) ?
793 dstricmp_sort(c1->name(), c2->name())<0 :
794 dstricmp_sort(c1->className(), c2->className())<0;
795 };
796 std::stable_sort(m_classes.begin(), m_classes.end(), classComp);
797
798 auto conceptComp = [](const ConceptLinkedRefMap::Ptr &c1,const ConceptLinkedRefMap::Ptr &c2)
799 {
800 return Config_getBool(SORT_BY_SCOPE_NAME) ?
801 dstricmp_sort(c1->qualifiedName(), c2->qualifiedName())<0 :
802 dstricmp_sort(c1->name(), c2->name())<0;
803 };
804 std::stable_sort(m_concepts.begin(), m_concepts.end(), conceptComp);
805 }
806
807 static auto contrComp = [](const ModuleDef *m1, const ModuleDef *m2)
808 {
809 FileDef *f1 = m1->getFileDef();
810 FileDef *f2 = m2->getFileDef();
811 DString fn1 = f1 ? f1->name() : m1->name();
812 DString fn2 = f2 ? f2->name() : m2->name();
813 static auto typeRank = [](const ModuleDef *m) -> int
814 {
815 if (m->moduleType()==ModuleDef::Type::Interface)
816 {
817 if (m->partitionName().empty()) return 0; // primary interface unit
818 return 1; // partition interface unit
819 }
820 else
821 {
822 if (!m->partitionName().empty()) return 2; // partition implementation unit
823 return 3; // implementation unit
824 }
825 };
826 auto tr1 = typeRank(m1);
827 auto tr2 = typeRank(m2);
828 int diff = dstricmp_sort(fn1,fn2);
829 return tr1<tr2 || (tr1==tr2 && diff<0);
830 };
831
832 std::stable_sort(m_contributing.begin(), m_contributing.end(), contrComp);
833}

1119 tagFile << " <compound kind=\"module\">\n";
1120 tagFile << " <name>" << convertToXML(name()) << "</name>\n";

1122 DString fn = fd ? fd->getOutputFileBase() : getOutputFileBase();
1123 addHtmlExtensionIfMissing(fn);
1124 tagFile << " <filename>" << convertToXML(fn) << "</filename>\n";
1125#if 0 // at the moment we do not export the members of a module to a tag file.
1126 // We let the project using a tag file directly link to the implementation of the
1127 // symbols (which have the same scope).

1129 // When we support linking to a module's interface instead we need to
1130 // export the module's members as well. Then we probably need a way to

1391void ModuleManager::resolvePartitionsRecursively(ModuleDef *intfMod, ModuleDef *mod)
1392{
1393 AUTO_TRACE();
1394 for (auto &[partitionFileName,importInfoList] : mod->getImports()) // foreach import
1395 {
1396 for (auto &importInfo : importInfoList)
1397 {
1398 AUTO_TRACE_ADD("partitionFileName={} importName={} partitionName={}",
1399 partitionFileName,importInfo.importName,importInfo.partitionName);
1400 if (importInfo.importName==intfMod->name() && !importInfo.partitionName.empty() &&
1401 importInfo.exported) // that is an exported partition of this module
1402 {
1403 auto it = p->moduleNameMap.find(importInfo.importName.str());
1404 if (it != p->moduleNameMap.end())
1405 {
1406 for (auto importedMod : it->second)
1407 {
1408 if (importedMod->qualifiedName()==importInfo.importName+":"+importInfo.partitionName)
1409 {
1410 AUTO_TRACE_ADD("Interface module {} exports partition {}:{}",
1411 mod->name(),importedMod->name(),importedMod->partitionName());
1412 toModuleDefImpl(intfMod)->addPartition(toModuleDefImpl(importedMod));
1413 toModuleDefImpl(importedMod)->setPrimaryInterface(intfMod);
1414 for (const auto &[partitionFileName_,partitionImportInfoList] : importedMod->getImports())
1415 {
1416 for (const auto &partitionImportInfo : partitionImportInfoList)
1417 {
1418 if (partitionImportInfo.exported && intfMod->name()!=partitionImportInfo.importName)
1419 {
1420 toModuleDefImpl(intfMod)->addExportedModule(partitionImportInfo.importName,partitionImportInfo);
1421 }
1422 }
1423 }
1424 resolvePartitionsRecursively(intfMod,importedMod);
1425 }
1426 }
1427 }
1428 }
1429 }
1430 }

1437 {
1438 if (mod->moduleType()==ModuleDef::Type::Interface && mod->partitionName().empty())
1439 { // that is a primary interface
1440 resolvePartitionsRecursively(mod.get(),mod.get());
1441 }
1442
1443 // copy exported imports to m_exportedModules
1444 for (const auto &[fileName,importInfoList] : mod->getImports())
1445 {
1446 for (const auto &importInfo : importInfoList)
1447 {
1448 if (importInfo.exported && mod->name()!=importInfo.importName)
1449 {
1450 toModuleDefImpl(mod)->addExportedModule(importInfo.importName,importInfo);
1451 }
1452 }
1453 }
1454
1455 // also link the ModuleDef and FileDef together
1456 bool ambig = false;
1457 FileDef *fd = Doxygen::inputNameLinkedMap->findFileDef(mod->getDefFileName(),ambig);
1458 if (fd)
1459 {
1460 fd->setModuleDef(mod.get());
1461 toModuleDefImpl(mod)->setFileDef(fd);
1462 }

1473 {
1474 for (const auto &[fileName,importInfoList] : mod->getImports())
